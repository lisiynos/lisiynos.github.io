• Оценка сложности

Оценка сложности алгоритмов: линейная сложность, квадратичная, логарифмическая...

Существует несколько способов измерения сложности алгоритма. Программисты обычно сосредотачивают внимание на скорости алгоритма, но не менее важны и другие показатели – требования к объёму памяти, например. Использование быстрого алгоритма не приведёт к ожидаемым результатам, если для его работы понадобится больше памяти, чем есть у компьютера.

При сравнении различных алгоритмов важно знать, как их сложность зависит от объёма входных данных.

$T(n): N -> Q^+ (>0)$

$T(n)=O(f(n))$ <=> сущ. C,N_0>0: T(n) \leq C*f(n), \all n \geq n_0

Лемма

$T(n) = a_0 * x^b + a_1 * x^{b-1} + ... + a_{n-1}*x^{1} + a_n*x^{0}$ $T(n) \leq c*x^b$ $n_0 = 1$ $c = |a_0| + |a_1| + ... $

Лемма 2

$\all k \geq 1, n^k \neq O(n^{k-1})$ \let (пусть) n^k = O(n^{k-1}) => \exist )C, $n^k \leq C * n^{k-1}$ $n \leq C, \forall n \geq n_0$

Худший случай..

$T(n) = \omega (f(n)) <=> \exist c,n_0 > 0: T(n) \geq c*f(n), \forall n \geq n.

$T(n) = \theta(f(n))$ \Leftrightarrow $T(n) = O(f(n))$ и $T(n) = \omega(f(n))$

$\exists c_1, c_2, n_0: C_1*f(n) \leq T(n) \leq C_2 * f(n), \forall n > n_0$

$\forall k > 0$ ...

Маленькая $o$

$T(n) = o(f(n))$ <=> $\forall c>0 \exists n_0:$ $T(n) \leq C*f(n)$ $\forall n \geq n_0$

Пример: $\forall k \geq 1$, $n^{k-1} = o(n^k)$

$O(1)$ - время выполнения алгоритма константа (не зависит от количества элементов во входном множестве).

$O(n)$ - линейная сложность.

Пример: поиск минимального элемента в массиве (цикл по всем элементам массива).

$O(n^2)$ - квадратичная сложность. Например, сортировки "пузырьком", вставками и т.д.

Реализация на Python

# Сортировка пузырьком: O(n^2) сравнений в среднем и в худшем случае
def bubble_sort(items):
    """Сортирует список по возрастанию на месте и возвращает его."""
    for last in range(len(items) - 1, 0, -1):
        # Самый большой из непросмотренных элементов «всплывает» в конец отрезка [0..last]
        for i in range(last):
            if items[i] > items[i + 1]:
                items[i], items[i + 1] = items[i + 1], items[i]
    return items


def main():
    print(bubble_sort([5, 3, 8, 1, 9, 2, 7]))


if __name__ == "__main__":
    main()

Реализация на Rust


// Сортировка пузырьком: O(n^2) сравнений в среднем и в худшем случае
pub fn bubble_sort(items: &mut [i32]) {
    for last in (1..items.len()).rev() {
        // Самый большой из непросмотренных элементов «всплывает» в конец отрезка [0..=last]
        for i in 0..last {
            if items[i] > items[i + 1] {
                items.swap(i, i + 1);
            }
        }
    }
}

fn main() {
    let mut items = vec![5, 3, 8, 1, 9, 2, 7];
    bubble_sort(&mut items);
    println!("{:?}", items);
}

Реализация на C++

// Сортировка пузырьком: O(n^2) сравнений в среднем и в худшем случае
#include <cstddef>
#include <iostream>
#include <utility>
#include <vector>

void bubble_sort(std::vector<int>& items) {
    // Самый большой из непросмотренных элементов «всплывает» в конец отрезка [0, last)
    for (std::size_t last = items.size(); last > 1; --last) {
        for (std::size_t i = 0; i + 1 < last; ++i) {
            if (items[i] > items[i + 1]) {
                std::swap(items[i], items[i + 1]);
            }
        }
    }
}

int main() {
    std::vector<int> items = {5, 3, 8, 1, 9, 2, 7};
    bubble_sort(items);
    for (int item : items) {
        std::cout << item << ' ';
    }
    std::cout << '\n';
    return 0;
}

Пример: поиск двух точек на плоскости с максимальным расстоянием между ними.

$O(n^3)$ - кубическая сложность.

Пример: поиск треугольника максимальной площади с вершинами в точках на плоскости (перебор вершин тройным циклом).

Оценка $O(n)$ ассимптотическая (при $n$ стремящемся к бесконечности).

$O(\ln{n})$ - Бинарный поиск - логарифмическая сложность.

Реализация на Python

# Бинарный поиск в отсортированном списке: O(log n) сравнений
def binary_search(items, target):
    """Возвращает индекс target в отсортированном списке или None, если его там нет."""
    low, high = 0, len(items)  # искомое ищем в полуинтервале [low, high)
    while low < high:
        middle = (low + high) // 2
        if items[middle] == target:
            return middle
        if items[middle] < target:
            low = middle + 1
        else:
            high = middle
    return None


def main():
    values = [1, 3, 5, 7, 9, 11, 13]
    print(binary_search(values, 7))
    print(binary_search(values, 8))


if __name__ == "__main__":
    main()

Реализация на Rust


// Бинарный поиск в отсортированном срезе: O(log n) сравнений
use std::cmp::Ordering;

pub fn binary_search(items: &[i32], target: i32) -> Option<usize> {
    // Искомое ищем в полуинтервале [low, high)
    let (mut low, mut high) = (0, items.len());
    while low < high {
        let middle = low + (high - low) / 2;
        match items[middle].cmp(&target) {
            Ordering::Equal => return Some(middle),
            Ordering::Less => low = middle + 1,
            Ordering::Greater => high = middle,
        }
    }
    None
}

fn main() {
    let values = [1, 3, 5, 7, 9, 11, 13];
    println!("{:?}", binary_search(&values, 7));
    println!("{:?}", binary_search(&values, 8));
}

Реализация на C++

// Бинарный поиск в отсортированном векторе: O(log n) сравнений
#include <cstddef>
#include <iostream>
#include <optional>
#include <vector>

std::optional<std::size_t> binary_search(const std::vector<int>& items, int target) {
    // Искомое ищем в полуинтервале [low, high)
    std::size_t low = 0;
    std::size_t high = items.size();
    while (low < high) {
        std::size_t middle = low + (high - low) / 2;
        if (items[middle] == target) {
            return middle;
        }
        if (items[middle] < target) {
            low = middle + 1;
        } else {
            high = middle;
        }
    }
    return std::nullopt;
}

int main() {
    const std::vector<int> values = {1, 3, 5, 7, 9, 11, 13};
    for (int target : {7, 8}) {
        auto index = binary_search(values, target);
        if (index) {
            std::cout << "индекс " << *index << '\n';
        } else {
            std::cout << "нет\n";
        }
    }
    return 0;
}

$O(n\log{n})$ - быстрая сортировка Хоара (рус) - QuickSort, HeapSort, MergeSort.

Реализация на Python

# Быстрая сортировка Хоара: O(n log n) в среднем, O(n^2) в худшем случае
def quick_sort(items):
    """Сортирует список по возрастанию на месте и возвращает его."""
    sort_range(items, 0, len(items) - 1)
    return items


def sort_range(items, low, high):
    """Сортирует отрезок items[low..high] (границы включены) разбиением Хоара."""
    if low >= high:
        return
    pivot = items[(low + high) // 2]
    i, j = low, high
    while i <= j:
        while items[i] < pivot:
            i += 1
        while items[j] > pivot:
            j -= 1
        if i <= j:
            items[i], items[j] = items[j], items[i]
            i += 1
            j -= 1
    # После разбиения [low..j] не больше опорного, [i..high] не меньше опорного
    sort_range(items, low, j)
    sort_range(items, i, high)


def main():
    print(quick_sort([5, 3, 8, 1, 9, 2, 7, 3]))


if __name__ == "__main__":
    main()

Реализация на Rust


// Быстрая сортировка Хоара: O(n log n) в среднем, O(n^2) в худшем случае
pub fn quick_sort(items: &mut [i32]) {
    if items.len() <= 1 {
        return;
    }
    let pivot = items[items.len() / 2];
    // Индексы со знаком: j может стать -1, и это не должно переполнять usize
    let (mut i, mut j) = (0_isize, items.len() as isize - 1);
    while i <= j {
        while items[i as usize] < pivot {
            i += 1;
        }
        while items[j as usize] > pivot {
            j -= 1;
        }
        if i <= j {
            items.swap(i as usize, j as usize);
            i += 1;
            j -= 1;
        }
    }
    // После разбиения [0..=j] не больше опорного, [i..] не меньше опорного
    if j >= 0 {
        quick_sort(&mut items[..=j as usize]);
    }
    quick_sort(&mut items[i as usize..]);
}

fn main() {
    let mut items = [5, 3, 8, 1, 9, 2, 7, 3];
    quick_sort(&mut items);
    println!("{:?}", items);
}

Реализация на C++

// Быстрая сортировка Хоара: O(n log n) в среднем, O(n^2) в худшем случае
#include <iostream>
#include <utility>
#include <vector>

// Сортирует отрезок items[low..high] (границы включены) разбиением Хоара
void sort_range(std::vector<int>& items, int low, int high) {
    if (low >= high) {
        return;
    }
    int pivot = items[(low + high) / 2];
    int i = low;
    int j = high;
    while (i <= j) {
        while (items[i] < pivot) {
            ++i;
        }
        while (items[j] > pivot) {
            --j;
        }
        if (i <= j) {
            std::swap(items[i], items[j]);
            ++i;
            --j;
        }
    }
    // После разбиения [low..j] не больше опорного, [i..high] не меньше опорного
    sort_range(items, low, j);
    sort_range(items, i, high);
}

void quick_sort(std::vector<int>& items) {
    sort_range(items, 0, static_cast<int>(items.size()) - 1);
}

int main() {
    std::vector<int> items = {5, 3, 8, 1, 9, 2, 7, 3};
    quick_sort(items);
    for (int item : items) {
        std::cout << item << ' ';
    }
    std::cout << '\n';
    return 0;
}

При оценке сложности важна только самая старшая степень. Например:

$O(n^3+n^2+4n)=O(n^3)$

Константа так же отбрасывается, так как нам важно лишь во сколько раз возрастает время выполнения программы при увеличении размера входных данных.

Разбор задач, день 3

Наибольшее расстояние

В некоторой стране $N$ городов и $M$ дорог. Каждая дорога представляет собой отрезок прямой, соединяющий два различных города, причём известно, что ни на одной дороге нет других городов, кроме тех, что задают её концы. Расстоянием между двумя городами считается минимальная суммарная длина дорог, по которым надо пройти, чтобы попасть из одного города в другой. Требуется вычислить наибольшее из таких расстояний.

# Открываем входной и выходной файлы
import sys
sys.stdin = open("maxdist.in") 
sys.stdout = open("maxdist.out", "w")

N = int(input()) # Количество городов
Inf = 1e20 # Обозначение для бесконечности
A = [[Inf]*N for i in range(N)] # Создаём матрицу A размера NxN 
for i in range(N): A[i][i] = 0  # Заполняем главную диагональ нулями 

C = [list(map(int,input().split())) for i in range(N)] # Считываем координаты городов

# Вводим дороги и считаем их длины
from math import sqrt 

M = int(input())
for k in range(M):
  i,j = map(int,input().split())
  i -= 1
  j -= 1
  A[i][j] = sqrt( (C[i][0]-C[j][0])**2 + (C[i][1]-C[j][1])**2 )
  A[j][i] = A[i][j]

# Алгоритм Флойда-Уоршалла
for k in range(N):
  for i in range(N):
    for j in range(N):
      t = A[i][k] + A[k][j]
      if t < A[i][j]:
        A[i][j] = t

# Выбираем максимальное из расстояний
maxdist = max(map(max,A))

# Выводим ответ
print(-1 if maxdist == Inf else float('%.12g' % maxdist))

Реализация на C++

// Алгоритм Флойда-Уоршелла: кратчайшие расстояния между всеми парами городов, O(N^3) времени
#include <algorithm>
#include <cstdio>
#include <limits>
#include <vector>

struct Road {
    int from;
    int to;
    double length;
};

using Matrix = std::vector<std::vector<double>>;

const double kInf = std::numeric_limits<double>::infinity();

// Для каждой промежуточной вершины k проверяем, не короче ли путь i -> k -> j
void floydWarshall(Matrix& dist) {
    const int n = static_cast<int>(dist.size());
    for (int k = 0; k < n; ++k) {
        for (int i = 0; i < n; ++i) {
            for (int j = 0; j < n; ++j) {
                dist[i][j] = std::min(dist[i][j], dist[i][k] + dist[k][j]);
            }
        }
    }
}

// Наибольшее из кратчайших расстояний между городами; -1, если какие-то города не связаны
double maxDistance(int n, const std::vector<Road>& roads) {
    Matrix dist(n, std::vector<double>(n, kInf));
    for (int i = 0; i < n; ++i) dist[i][i] = 0;
    for (const Road& road : roads) {
        dist[road.from][road.to] = std::min(dist[road.from][road.to], road.length);
        dist[road.to][road.from] = dist[road.from][road.to];
    }
    floydWarshall(dist);
    double best = 0;
    for (const std::vector<double>& row : dist) {
        for (double d : row) {
            if (d == kInf) return -1;
            best = std::max(best, d);
        }
    }
    return best;
}

int main() {
    // Города 0..3 и дороги (откуда, куда, длина)
    const std::vector<Road> roads = {{0, 1, 3}, {1, 2, 4}, {2, 3, 2}, {3, 0, 6}, {0, 2, 10}};
    std::printf("Наибольшее расстояние = %.0f\n", maxDistance(4, roads));
    return 0;
}

Реализация на Rust

// Алгоритм Флойда-Уоршелла: кратчайшие расстояния между всеми парами городов, O(N^3) времени
// Для каждой промежуточной вершины k проверяем, не короче ли путь i -> k -> j
pub fn floyd_warshall(dist: &mut [Vec<f64>]) {
    let n = dist.len();
    for k in 0..n {
        for i in 0..n {
            for j in 0..n {
                dist[i][j] = dist[i][j].min(dist[i][k] + dist[k][j]);
            }
        }
    }
}

// Наибольшее из кратчайших расстояний между городами; None, если какие-то города не связаны
pub fn max_distance(n: usize, roads: &[(usize, usize, f64)]) -> Option<f64> {
    let mut dist = vec![vec![f64::INFINITY; n]; n];
    for (i, row) in dist.iter_mut().enumerate() {
        row[i] = 0.0;
    }
    for &(from, to, length) in roads {
        let shortest = dist[from][to].min(length);
        dist[from][to] = shortest;
        dist[to][from] = shortest;
    }
    floyd_warshall(&mut dist);
    let mut best = 0.0_f64;
    for &d in dist.iter().flatten() {
        if d.is_infinite() {
            return None;
        }
        best = best.max(d);
    }
    Some(best)
}

fn main() {
    // Города 0..3 и дороги (откуда, куда, длина)
    let roads = [(0, 1, 3.0), (1, 2, 4.0), (2, 3, 2.0), (3, 0, 6.0), (0, 2, 10.0)];
    match max_distance(4, &roads) {
        Some(d) => println!("Наибольшее расстояние = {d}"),
        None => println!("Есть несвязанные города"),
    }
}

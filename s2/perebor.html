Перебор вариантов

Очень часто можно слышать фразу «ЭВМ проанализировала огромное количество вариантов и выдала наилучший» [4]. Так обычно говорят о тех задачах, которые требуют при поиске решения перебора вариантов.

  • Простейшие задачи на перебор – это задачи на поиск решения в одномерном массиве (например, найти элемент с заданным свойством). Сложность таких задач пропорциональна количеству элементов в массиве $N$.
  • Следующие задачи на перебор – поиск пар элементов либо из одного массива, либо из двух разных массивов. При решении таких задач, как правило, используется вложенный цикл, то есть их сложность пропорциональна $N^2$ .
  • Далее идет перебор троек элементов. Рассмотрим, например, следующую задачу.

Треугольник максимальной площади

На плоскости разбросаны $N$ точек с координатами $(x_1, y_1), (x_2, y_2), ..., (x_N, y_N)$. Найти тройку точек, которые образуют треугольник с максимальной площадью.

Ясно, что при решении этой задачи необходимо использовать три вложенных цикла, таким образом, задача решается за $N^3$ шагов.

Для задания координат точек используем датчик случайных чисел.

Реализация на Python

from random import *

N = 100
MIN_X = -10000
MAX_X = 10000
MIN_Y = -10000
MAX_Y = 10000
# Генерируем N случайных точек
points = [(randrange(MIN_X, MAX_X), randrange(MIN_Y, MAX_Y)) for i in range(N)]


def area(p1, p2, p3):
    """ Удвоенная площадь треугольника (модуль векторного произведения) """
    x1, y1 = p1
    x2, y2 = p2
    x3, y3 = p3
    # Получаем вектора
    ax, ay = x2 - x1, y2 - y1
    bx, by = x3 - x1, y3 - y1
    # Векторное произведение
    cross_product = ax * by - ay * bx
    # Возвращаем удвоенную площадь (целое число)
    return abs(cross_product)


# Проверяем работу функции
assert area((0, 0), (0, 1), (1, 0)) == 1
assert area((1, 1), (1, 2), (3, 1)) == 2

S_max = 0
triangle = []  # Номера вершин треугольника с максимальной площадью

# Начинаем перебор: перебираем тройки точек
for i in range(N):
    for j in range(i + 1, N):
        for k in range(j + 1, N):
            S = area(points[i], points[j], points[k])
            if S > S_max:
                S_max = S
                # Запоминаем номера точек
                triangle = [i + 1, j + 1, k + 1]

print("Удвоенная площадь =", S_max)
if S_max > 0:
    print("Номера точек:", *triangle)

Реализация на C++

// Перебор троек точек: треугольник максимальной площади, O(N^3) времени
#include <cstdio>
#include <vector>

struct Point {
    long long x;
    long long y;
};

// Удвоенная площадь треугольника abc: модуль векторного произведения (целое число)
long long doubledArea(const Point& a, const Point& b, const Point& c) {
    long long cross = (b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x);
    return cross < 0 ? -cross : cross;
}

struct BestTriangle {
    long long area2 = 0;         // удвоенная площадь лучшего треугольника
    int i = -1, j = -1, k = -1;  // номера его вершин (от 0)
};

// Перебираем все тройки i < j < k и запоминаем треугольник наибольшей площади
BestTriangle maxTriangle(const std::vector<Point>& points) {
    BestTriangle best;
    const int n = static_cast<int>(points.size());
    for (int i = 0; i < n; ++i) {
        for (int j = i + 1; j < n; ++j) {
            for (int k = j + 1; k < n; ++k) {
                long long area2 = doubledArea(points[i], points[j], points[k]);
                if (area2 > best.area2) best = {area2, i, j, k};
            }
        }
    }
    return best;
}

int main() {
    const std::vector<Point> points = {{0, 0}, {4, 0}, {0, 3}, {1, 1}, {5, 5}, {-2, 4}, {3, -3}};
    const BestTriangle best = maxTriangle(points);
    std::printf("Удвоенная площадь = %lld\n", best.area2);
    std::printf("Номера точек: %d %d %d\n", best.i + 1, best.j + 1, best.k + 1);
    return 0;
}

Реализация на Rust

// Перебор троек точек: треугольник максимальной площади, O(N^3) времени
#[derive(Clone, Copy)]
pub struct Point {
    pub x: i64,
    pub y: i64,
}

// Удвоенная площадь треугольника abc: модуль векторного произведения (целое число)
pub fn doubled_area(a: Point, b: Point, c: Point) -> i64 {
    ((b.x - a.x) * (c.y - a.y) - (b.y - a.y) * (c.x - a.x)).abs()
}

// Перебираем все тройки i < j < k и запоминаем треугольник наибольшей площади.
// Возвращает удвоенную площадь и номера вершин (от 0); None, если все площади нулевые.
pub fn max_triangle(points: &[Point]) -> (i64, Option<(usize, usize, usize)>) {
    let n = points.len();
    let mut best = (0, None);
    for i in 0..n {
        for j in i + 1..n {
            for k in j + 1..n {
                let area2 = doubled_area(points[i], points[j], points[k]);
                if area2 > best.0 {
                    best = (area2, Some((i, j, k)));
                }
            }
        }
    }
    best
}

fn main() {
    let points = [
        Point { x: 0, y: 0 },
        Point { x: 4, y: 0 },
        Point { x: 0, y: 3 },
        Point { x: 1, y: 1 },
        Point { x: 5, y: 5 },
        Point { x: -2, y: 4 },
        Point { x: 3, y: -3 },
    ];
    let (area2, triple) = max_triangle(&points);
    println!("Удвоенная площадь = {area2}");
    if let Some((i, j, k)) = triple {
        println!("Номера точек: {} {} {}", i + 1, j + 1, k + 1);
    }
}

Перебор с отсечениями

Решить уравнение в целых числах: $x_1^4 + x_2^4 + ... + x_{15}^4 = 2000$

Чтобы ответ был единственным, добавим условие: $x_i \ge x_{i-1}$

Реализация на Python

x = [0] * 16  # значения переменных x, x[0] всегда равно 0


def rec(i, s):  # рекурсивная процедура
    """ i - индекс икса, s - оставшаяся сумма """
    if i == 16:  # Получили набор x
        if s == 0:  # набрана нужная сумма
            # Выводим ответ
            print(*x[1:])
            # Для проверки вычисляем сумму
            print(sum([v ** 4 for v in x]))
        return
    if s < 0:  # Если оставшаяся сумма отрицательна => на выход
        return
    # Перебираем значение x от значения предыдущего икса до
    # корня четвёртой степени из оставшейся суммы
    for value in range(x[i - 1], int(s ** 0.25) + 1):
        x[i] = value  # фиксируем значение x[i]
        rec(i + 1, s - value ** 4)

# Запускаем перебор
rec(1, 2000)
# Получаем: 2 2 2 2 2 2 2 2 2 2 2 2 4 4 6

Приведенная программа ищет решения уравнения $x_1^4 + x_2^4 + ... + x_{15}^4 = 2000$ в натуральных числах методом перебора. Перебор является, пожалуй, одним из самых простых (по реализации), но, вместе с тем, самым медленным (по времени выполнения) из всех алгоритмов, поэтому при решении каждой конкретной задачи перебором возникает другая задача – сократить и оптимизировать перебор, иными словами, максимально уменьшить время, затрачиваемое компьютером на поиск решения.

Познакомимся с некоторыми возможными приемами сокращения перебора на данной конкретной задаче.

1) В первую очередь, надо правильно выбрать границы значений перебираемых величин. По условию задачи все переменные принимают натуральные значения, но ведь натуральных чисел бесконечно много! Однако, в данном примере совершенно очевидно, что, если решение уравнения существует, то все $x_1, ..., x_{15}$ лежат в промежутке от 1 до 6, поскольку натуральные числа, большие 6, в четвертой степени дают значение уже превосходящее 2000 ($7^4 = 2401 \gt 2000$). Итак, в алгоритме все переменные перебираются в пределах от 1 до 6.

2) Далее, важным моментом при оптимизации является отсечение заведомо неверных ветвей перебора. Например, в данном примере, не имеет смысла продолжать поиск, если на некотором шаге сумма $x_1^4 + ... + x_n^4$ уже превысила 2000.

3) Следует избегать многократного выполнения одних и тех же операций в циклах и рекурсивных процедурах, если данные, используемые в этих циклах, могут быть подготовлены заранее. Так, в приведенном примере значения четвертых степеней чисел 1, 2, …, 6 вычислены один раз и занесены в массив констант: p4 = (1, 16, 81, 256, 625, 1296). Подобный прием часто помогает избавиться от лишних вычислений, например, в алгоритмах, работающих с простыми числами, полезно бывает включить заранее найденные простые числа в текст программы, так же, в виде массива констант.

4) Кроме того, можно существенно ускорить перебор, если не рассматривать варианты, аналогичные (либо похожие) ранее рассмотренным. Так, в данной задаче решения уравнения определены с точностью до перестановки слагаемых, т.е., если найдено одно решение, то любая перестановка переменных в нем тоже будет являться решением. Поэтому, чтобы сократить поиск, не перебирая варианты, отличающиеся только перестановкой слагаемых, в процедуру Find был введен дополнительный параметр start – максимальное значение, которое может принимать очередная ($N$-ая) переменная. Благодаря этому гарантируется, что алгоритм будет рассматривать только те значения переменных, при которых $x_1$ ³ $x_2$ ³ … ³ $x_{15}$ .

5) Наконец, если задача предполагает поиск какого-либо одного решения, а не всех, можно добиться более быстрого получения результата путем выбора хорошего начального приближения, либо перебором с предпочтением (так называемыми эвристическими методами). Эти методы различны для каждого конкретного случая, но, как правило, заключаются в том или ином способе реализации «жадного» алгоритма. В указанной программе эвристика заключается в том, что перебор осуществляется от большего значения к меньшему, а не наоборот: for i=start to 1 step -1. Интуитивно понятно, что, начиная перебор с больших значений переменных (6, 5, 4), мы быстрее «доберемся» до решения нежели, чем «пробуя» меньшие значения.

Реализация на C++

// Перебор с отсечениями: x1^4 + ... + xn^4 = S (натуральные, x1 <= ... <= xn), без отсечений O(S^(n/4))
#include <cstdio>
#include <vector>

using Solution = std::vector<long long>;

// Подбирает countLeft чисел не меньше low с суммой четвертых степеней rest
void search(int countLeft, long long low, long long rest, Solution& chosen,
            std::vector<Solution>& solutions) {
    if (countLeft == 0) {
        if (rest == 0) solutions.push_back(chosen);
        return;
    }
    // Отсечение: даже countLeft одинаковых чисел value дают сумму больше rest
    for (long long value = low; countLeft * value * value * value * value <= rest; ++value) {
        chosen.push_back(value);
        search(countLeft - 1, value, rest - value * value * value * value, chosen, solutions);
        chosen.pop_back();  // возврат
    }
}

// Все наборы натуральных чисел x1 <= ... <= xn с суммой четвертых степеней target
std::vector<Solution> findSolutions(int n, long long target) {
    std::vector<Solution> solutions;
    Solution chosen;
    search(n, 1, target, chosen, solutions);
    return solutions;
}

int main() {
    const std::vector<Solution> solutions = findSolutions(15, 2000);
    std::printf("Решений: %zu\n", solutions.size());
    for (const Solution& solution : solutions) {
        for (long long value : solution) std::printf("%lld ", value);
        std::printf("\n");
    }
    return 0;
}

// Перебор с отсечениями: x1^4 + ... + xn^4 = S (натуральные, x1 <= ... <= xn), без отсечений O(S^(n/4))

// Подбирает count_left чисел не меньше low с суммой четвертых степеней rest
pub fn search(
    count_left: u32,
    low: u64,
    rest: u64,
    chosen: &mut Vec<u64>,
    solutions: &mut Vec<Vec<u64>>,
) {
    if count_left == 0 {
        if rest == 0 {
            solutions.push(chosen.clone());
        }
        return;
    }
    let mut value = low;
    // Отсечение: даже count_left одинаковых чисел value дают сумму больше rest
    while u64::from(count_left) * value.pow(4) <= rest {
        chosen.push(value);
        search(count_left - 1, value, rest - value.pow(4), chosen, solutions);
        chosen.pop(); // возврат
        value += 1;
    }
}

// Все наборы натуральных чисел x1 <= ... <= xn с суммой четвертых степеней target
pub fn find_solutions(n: u32, target: u64) -> Vec<Vec<u64>> {
    let mut solutions = Vec::new();
    search(n, 1, target, &mut Vec::new(), &mut solutions);
    solutions
}

fn main() {
    let solutions = find_solutions(15, 2000);
    println!("Решений: {}", solutions.len());
    for solution in &solutions {
        let line: Vec<String> = solution.iter().map(|v| v.to_string()).collect();
        println!("{}", line.join(" "));
    }
}

# Перебор с отсечениями: ребус word+...+word = total (k слагаемых), O(10^L) вариантов, L - длина слова
# Буквы записаны латиницей: МУХА - MUHA, СЛОН - SLON. Разным буквам соответствуют разные цифры.


def match(mapping, letters, number):
    """Дописывает к mapping соответствие букв цифрам числа; False, если буква уже означает другую цифру."""
    for letter, digit in zip(letters, map(int, str(number))):
        if letter in mapping and mapping[letter] != digit:
            return False
        mapping[letter] = digit
    return True


def solve(word, total, k):
    """Все решения ребуса word+...+word (k слагаемых) = total в виде словарей {буква: цифра}."""
    if not word or not total:
        return []
    solutions = []
    for number in range(10 ** (len(word) - 1), 10 ** len(word)):  # число без ведущего нуля
        product = k * number
        if product == 0 or len(str(product)) != len(total):  # отсечение: ноль или сумма другой длины
            continue
        mapping = {}
        if not (match(mapping, word, number) and match(mapping, total, product)):
            continue
        if len(set(mapping.values())) == len(mapping):  # разным буквам - разные цифры
            solutions.append(mapping)
    return solutions


def main():
    solutions = solve("MUHA", "SLON", 3)
    print("Решений:", len(solutions))
    for mapping in solutions[:3]:
        muha = "".join(str(mapping[c]) for c in "MUHA")
        slon = "".join(str(mapping[c]) for c in "SLON")
        print(f"MUHA={muha} SLON={slon}")


if __name__ == "__main__":
    main()

Реализация на C++

// Перебор с отсечениями: ребус word+...+word = total (k слагаемых), O(10^L) вариантов, L - длина слова
// Буквы записаны латиницей: МУХА - MUHA, СЛОН - SLON. Разным буквам соответствуют разные цифры.
#include <iostream>
#include <map>
#include <string>
#include <vector>

using Mapping = std::map<char, int>;  // буква -> цифра

// Дописывает к mapping соответствие букв цифрам числа; false, если буква уже означает другую цифру
bool match(Mapping& mapping, const std::string& letters, long long number) {
    std::string digits = std::to_string(number);
    for (size_t i = 0; i < letters.size(); ++i) {
        int digit = digits[i] - '0';
        auto found = mapping.find(letters[i]);
        if (found != mapping.end() && found->second != digit) return false;
        mapping[letters[i]] = digit;
    }
    return true;
}

// Все решения ребуса: словари {буква: цифра}
std::vector<Mapping> solve(const std::string& word, const std::string& total, long long k) {
    std::vector<Mapping> solutions;
    if (word.empty() || total.empty()) return solutions;
    long long low = 1;  // 10^(L-1): число без ведущего нуля
    for (size_t i = 1; i < word.size(); ++i) low *= 10;
    for (long long number = low; number < low * 10; ++number) {
        long long product = k * number;
        if (product == 0 || std::to_string(product).size() != total.size()) continue;  // отсечение
        Mapping mapping;
        if (!match(mapping, word, number) || !match(mapping, total, product)) continue;
        std::vector<bool> used(10, false);  // разным буквам - разные цифры
        bool distinct = true;
        for (const auto& entry : mapping) {
            if (used[entry.second]) distinct = false;
            used[entry.second] = true;
        }
        if (distinct) solutions.push_back(mapping);
    }
    return solutions;
}

int main() {
    std::vector<Mapping> solutions = solve("MUHA", "SLON", 3);
    std::cout << "Решений: " << solutions.size() << "\n";
    for (size_t s = 0; s < solutions.size() && s < 3; ++s) {
        std::cout << "MUHA=";
        for (char c : std::string("MUHA")) std::cout << solutions[s].at(c);
        std::cout << " SLON=";
        for (char c : std::string("SLON")) std::cout << solutions[s].at(c);
        std::cout << "\n";
    }
    return 0;
}

fn match_digits(mapping: &mut Mapping, letters: &str, number: u64) -> bool {
    let digits = number.to_string();
    for (letter, digit) in letters.chars().zip(digits.bytes()) {
        let digit = digit - b'0';
        if let Some(&known) = mapping.get(&letter) {
            if known != digit {
                return false;
            }
        }
        mapping.insert(letter, digit);
    }
    true
}

// Все решения ребуса: словари {буква: цифра}
pub fn solve(word: &str, total: &str, k: u64) -> Vec<Mapping> {
    let mut solutions = Vec::new();
    if word.is_empty() || total.is_empty() {
        return solutions;
    }
    let low = 10u64.pow(word.len() as u32 - 1); // 10^(L-1): число без ведущего нуля
    for number in low..low * 10 {
        let product = k * number;
        if product == 0 || product.to_string().len() != total.len() {
            continue; // отсечение: ноль или сумма другой длины
        }
        let mut mapping = Mapping::new();
        if !match_digits(&mut mapping, word, number) || !match_digits(&mut mapping, total, product) {
            continue;
        }
        let mut digits: Vec<u8> = mapping.values().copied().collect();
        digits.sort_unstable();
        digits.dedup();
        if digits.len() == mapping.len() {
            // разным буквам - разные цифры
            solutions.push(mapping);
        }
    }
    solutions
}

Данную задачу на математический язык можно перефразировать так: по заданной перестановке из семи элементов требуется сгенерировать следующую за ней в лексикографическом порядке. Так как порядок букв в алфавите Икнатсо отличен от обычного, лучше оперировать не с буквами, а с числами от 1 до 7. Сопоставим буквам И, К, Н, А, Т, С, О их порядок в алфавите: букву «И» обозначим единицей, «К» - двойкой, и т. д., таким образом, исходное слово представим в виде массива из семи чисел. Далее вместо слова из алфавита Икнатсо будем говорить об этом массиве.

Проще всего получить следующую по порядку перестановку, когда последний (седьмой) элемент массива больше предпоследнего. В этом случае, чтобы получить ответ, достаточно поменять местами эти два последних элемента. В противном случае, надо рассмотреть два предпоследних элемента (пятый и шестой). Если пятый оказался меньше, на его место нужно поставить тот из двух последних, который должен следовать за ним (т.е. тот, который больше него, но, в то же время, меньше всех остальных). Так, например, в массиве (…, 3, 6, 5) нужно поменять местами тройку с пятеркой, а в массиве (…, 4, 6, 2) – четверку с шестеркой. После этого два последних элемента нужно записать в порядке возрастания (т.е. по алфавиту). Таким образом, после этой операции первый массив преобразуется в (…, 5, 3, 6), а второй – в (…, 6, 2, 4). Если же пятый элемент оказался больше шестого, потребуется рассмотреть предыдущую пару (четвертый с пятым). И так далее, до самого первого элемента.

Итак, весь алгоритм заключается в следующем:

1.     Запишем исходное слово в виде массива чисел от 1 до 7. Назовем его A.

2.     Будем последовательно рассматривать пары элементов Ai, Ai+1 (i = 6, 5, …, 1), пока не найдем такую пару, где Ai < Ai+1. Если такой пары не найдется, это означает, что весь массив отсортирован в порядке убывания, т.е. следующей перестановки не существует, а введенное слово является в словаре Икнатсо последним.

3.     Пусть оказалось, что Ai < Ai+1. Среди элементов Ai+1, Ai+2, …, $A_7$ найдем наибольшй Aj, такой что Aj > Ai. Поменяем элементы Ai и Aj местами.

4.     Упорядочим элементы Ai+1, Ai+2, …, $A_7$ по возрастанию.

5.     Полученный массив преобразуем обратно в слово. Числу 1 сопоставим букву «И», числу 2 – «К», и т.д. Полученное слово – искомое.

2.3.2. Перебор с возвратом (backtracking)

Любая задача, к которой применим алгоритм перебора с возвратом (бектрекинг), может быть описана в общем случае следующим образом: требуется построить к данной задаче вектор решения (a1, a2, …, an), удовлетворяющий множеству условий и ограничений. Такой вектор строится покомпонентно слева направо. Предположим, что уже найдены значения первых k-1 компонент, а выбор следующей компоненты ak зависит от некоторых ограничений и условий. Если условия и ограничения выполнимы, то выбираем ak и переходим к рассмотрению компоненты ak+1, затем  ak+2 и так далее, если же оказалось невозможным выбрать компоненту ak, то необходимо вернуться к предыдущему этапу, выбросить (k-1)-й элемент и выбрать другой ak-1, перейдя после этого снова к выбору k-й компоненты. Такой перебор может возвращать назад на несколько шагов,  вплоть до выбора самой первой компоненты.

Решаемые бектрекингом задачи, как правило, принадлежат одному из трех классов – требуется найти либо произвольное из решений, либо перечислить все возможные решения, либо найти решение оптимальное по заданному критерию [2].

Программная реализация метода перебора с возвратом заключается в том, что на каждом шаге, перед рассмотрением очередного варианта, запоминается текущееперебора, а после анализа этого варианта – сохраненные значения восстанавливаются ири рассмотрении следующего. По окончании анализа некоторого варианта, данные, записанные последними, восстанавливаются в первую очередь, поэтому для хранения состояний удобно использовать структуры данных на основе стека. По этой же причине перебор с возвратом часто органрекурсивной процедуры.

В данной задаче под «состоянием перебора» подразумевасобранных в рюкзак предметов (Used), их суммарная стоимость (TotalCost) и общий вес (TotalWeight). Алгоритм перебора представлен рекурсивной процедурой, в которую в качестве параметра (Last) передается порядковый номер последнего добавленного в рюкзак пре самом начале, пока рюкзак пустой).  Процедура перебирает предметы с номерами, большими Last, и, если  iет может поместиться в рюкзак (т.е. его вес Weight(iс текущим весом рюкзака TotalWeight не превысит максимальную вместимость BagSize), то выполняет над ним следующие операции:

1.      «помещает» i-ый предмет в рюкзак, т.е. помечает в массиве Used этот предмет как использованный;

2.      пересчитывает общий вес и суммарную стоимость предметов рюкзака с учетом добавленного i-го предмета;

3.      в случае, когда общая стоимость оказалась больше ранее найденной максимальной стоимости предметов, текущий вариант заполнения рюкзака запаилучший;

4.      эта же процедура вызывается рекурсивно для рассмотрения оставшихся предметов с новым значением параметра Last, равным i;

5.      по возвращении из рекурсивной процедуры i-ый предмет «вынимается» из рюкзака (помечается как неиспользованный), и снова пересчитывается общая стоимость и вес предметов (восстанавливаются прежние значения, не учитывающие i-ый предмет);

6.      продолжается перебор вариа следующего, (i+1)-го предмета.

 

После завершения работы алгоритма перебора достаточно вывести на печать запомненный наилучший вариант заполнения рюкзака – это и будет являться ответом>

# Перебор с возвратом (backtracking): задача о рюкзаке, O(2^m) вариантов, отсечение по весу
def knapsack(weights, costs, capacity):
    """Возвращает (максимальная стоимость, номера предметов) для рюкзака вместимостью capacity."""
    best = (0, [])
    chosen = []  # предметы, которые сейчас лежат в рюкзаке

    def search(first, weight, cost):
        nonlocal best
        if cost > best[0]:
            best = (cost, list(chosen))
        for item in range(first, len(weights)):
            # Отсечение: предмет не помещается в рюкзак
            if weight + weights[item] > capacity:
                continue
            chosen.append(item)  # кладем предмет в рюкзак
            search(item + 1, weight + weights[item], cost + costs[item])
            chosen.pop()  # возврат: вынимаем предмет и пробуем следующий

    search(0, 0, 0)
    return best

// Перебор с возвратом (backtracking): задача о рюкзаке, O(2^m) вариантов, отсечение по весу
#include <cstdio>
#include <vector>

struct Problem {
    std::vector<long long> weights;
    std::vector<long long> costs;
    long long capacity;
};

struct Bag {
    long long cost = 0;
    std::vector<int> items;  // номера предметов (от 0), положенных в рюкзак
};

// Рекурсивно перебирает наборы предметов с номерами от first; текущий набор лежит в chosen
void search(const Problem& problem, int first, long long weight, long long cost,
            std::vector<int>& chosen, Bag& best) {
    if (cost > best.cost) best = {cost, chosen};
    const int count = static_cast<int>(problem.weights.size());
    for (int item = first; item < count; ++item) {
        // Отсечение: предмет не помещается в рюкзак
        if (weight + problem.weights[item] > problem.capacity) continue;
        chosen.push_back(item);  // кладем предмет
        search(problem, item + 1, weight + problem.weights[item], cost + problem.costs[item],
               chosen, best);
        chosen.pop_back();  // возврат: вынимаем предмет и пробуем следующий
    }
}

Bag knapsack(const Problem& problem) {
    Bag best;
    std::vector<int> chosen;
    search(problem, 0, 0, 0, chosen, best);
    return best;
}

int main() {
    const Problem problem{{3, 4, 2, 5, 9}, {10, 4, 7, 8, 13}, 10};
    const Bag best = knapsack(problem);
    std::printf("Максимальная стоимость: %lld\n", best.cost);
    std::printf("Номера предметов:");
    for (int item : best.items) std::printf(" %d", item + 1);
    std::printf("\n");
    return 0;
}

// Перебор с возвратом (backtracking): задача о рюкзаке, O(2^m) вариантов, отсечение по весу

    pub weights: Vec<u64>,
    pub costs: Vec<u64>,
    pub capacity: u64,
}

pub struct Bag {
    pub cost: u64,
    pub items: Vec<usize>, // номера предметов (от 0), положенных в рюкзак
}

// Рекурсивно перебирает наборы предметов с номерами от first; текущий набор лежит в chosen
pub fn search(
    problem: &Problem,
    first: usize,
    weight: u64,
    cost: u64,
    chosen: &mut Vec<usize>,
    best: &mut Bag,
) {
    if cost > best.cost {
        *best = Bag { cost, items: chosen.clone() };
    }
    for item in first..problem.weights.len() {
        let new_weight = weight + problem.weights[item];
        // Отсечение: предмет не помещается в рюкзак
        if new_weight > problem.capacity {
            continue;
        }
        chosen.push(item); // кладем предмет
        search(problem, item + 1, new_weight, cost + problem.costs[item], chosen, best);
        chosen.pop(); // возврат: вынимаем предмет и пробуем следующий
    }
}

pub fn knapsack(problem: &Problem) -> Bag {
    let mut best = Bag { cost: 0, items: Vec::new() };
    search(problem, 0, 0, 0, &mut Vec::new(), &mut best);
    best
}

        costs: vec![10, 4, 7, 8, 13],

// Перебор с возвратом и эвристика Варнсдорфа: обход коня на доске n x n, O(8^(n*n)) в худшем случае
#include <algorithm>
#include <iomanip>
#include <iostream>
#include <optional>
#include <utility>
#include <vector>

using Board = std::vector<std::vector<int>>;  // номера ходов, 0 - клетка свободна
using Cell = std::pair<int, int>;

const int DR[8] = {1, 2, 2, 1, -1, -2, -2, -1};  // ходы коня (порядок такой же, как в Python)
const int DC[8] = {2, 1, -1, -2, -2, -1, 1, 2};

// Свободные клетки, куда конь может прыгнуть из клетки (row, col)
std::vector<Cell> free_moves(const Board& board, int row, int col) {
    int n = static_cast<int>(board.size());
    std::vector<Cell> cells;
    for (int m = 0; m < 8; ++m) {
        int r = row + DR[m], c = col + DC[m];
        if (r >= 0 && r < n && c >= 0 && c < n && board[r][c] == 0) cells.push_back({r, c});
    }
    return cells;
}

// Ставит номер хода на клетку (row, col) и продолжает обход; true, если обход найден
bool search(Board& board, int row, int col, int number) {
    board[row][col] = number;
    int n = static_cast<int>(board.size());
    if (number == n * n) return true;
    std::vector<Cell> cells = free_moves(board, row, col);
    // эвристика Варнсдорфа: сначала идем в клетку, из которой меньше всего ходов
    std::stable_sort(cells.begin(), cells.end(), [&board](Cell a, Cell b) {
        return free_moves(board, a.first, a.second).size() < free_moves(board, b.first, b.second).size();
    });
    for (Cell cell : cells) {
        if (search(board, cell.first, cell.second, number + 1)) return true;
    }
    board[row][col] = 0;  // тупик: убираем номер и возвращаемся назад
    return false;
}

// Номера ходов 1..n*n на доске n x n или nullopt, если обхода из стартовой клетки нет
std::optional<Board> knight_tour(int n, int start_row, int start_col) {
    if (start_row < 0 || start_row >= n || start_col < 0 || start_col >= n) return std::nullopt;
    Board board(n, std::vector<int>(n, 0));
    if (search(board, start_row, start_col, 1)) return board;
    return std::nullopt;
}

pub const MOVES: [(i32, i32); 8] = [(1, 2), (2, 1), (2, -1), (1, -2), (-1, -2), (-2, -1), (-2, 1), (-1, 2)];

        .iter()
        .map(|&(dr, dc)| (row as i32 + dr, col as i32 + dc))
        .filter(|&(r, c)| r >= 0 && r < n && c >= 0 && c < n && board[r as usize][c as usize] == 0)

        }

# Поиск в ширину (волна на графе): кратчайший путь от клетки (row, col) до границы лабиринта, O(N*M) времени
# В лабиринте 0 - проходимая клетка, 1 - стена; выход - любая проходимая граничная клетка
from collections import deque

DIRECTIONS = [(0, 1), (1, 0), (0, -1), (-1, 0)]

MAZE = [
    "####.####",
    "#.......#",
    "#.#####.#",
    "#.#...#.#",
    "#.#.#.#.#",
    "#...#...#",
    "#########",
]


def on_border(grid, row, col):
    return row in (0, len(grid) - 1) or col in (0, len(grid[0]) - 1)


def restore(parent, cell):
    """Восстанавливает путь от старта до клетки cell по ссылкам на предыдущие клетки."""
    path = []
    while cell is not None:
        path.append(cell)
        cell = parent[cell]
    return path[::-1]


def shortest_exit(maze, start):
    """Кратчайший путь от start до границы (список клеток) или None: клетки обходятся очередью."""
    if maze[start[0]][start[1]] == 1:
        return None
    parent = {start: None}
    queue = deque([start])
    while queue:
        row, col = queue.popleft()
        if on_border(maze, row, col):
            return restore(parent, (row, col))
        for dr, dc in DIRECTIONS:
            r, c = row + dr, col + dc
            if maze[r][c] == 0 and (r, c) not in parent:
                parent[(r, c)] = (row, col)
                queue.append((r, c))
    return None


def main():
    maze = [[int(ch == "#") for ch in line] for line in MAZE]
    path = shortest_exit(maze, (5, 1))
    if path is None:
        print("Выхода нет")
        return
    print("Шагов:", len(path) - 1)
    print(path)


if __name__ == "__main__":
    main()

// Поиск в ширину (волна на графе): кратчайший путь от клетки (row, col) до границы лабиринта, O(N*M) времени
// В лабиринте 0 - проходимая клетка, 1 - стена; выход - любая проходимая граничная клетка
#include <algorithm>
#include <iostream>
#include <map>
#include <queue>
#include <utility>
#include <vector>

using Grid = std::vector<std::vector<int>>;
using Cell = std::pair<int, int>;

const int DR[4] = {0, 1, 0, -1};
const int DC[4] = {1, 0, -1, 0};
const char* const MAZE[] = {"####.####", "#.......#", "#.#####.#", "#.#...#.#",
                            "#.#.#.#.#", "#...#...#", "#########"};

bool on_border(const Grid& grid, Cell cell) {
    return cell.first == 0 || cell.first == static_cast<int>(grid.size()) - 1 || cell.second == 0 ||
           cell.second == static_cast<int>(grid[0].size()) - 1;
}

// Кратчайший путь от start до границы (список клеток) или пустой список: клетки обходятся очередью
std::vector<Cell> shortest_exit(const Grid& maze, Cell start) {
    if (maze[start.first][start.second] == 1) return {};
    std::map<Cell, Cell> parent = {{start, start}};  // у старта предыдущая клетка - он сам
    std::queue<Cell> queue;
    queue.push(start);
    while (!queue.empty()) {
        Cell cell = queue.front();
        queue.pop();
        if (on_border(maze, cell)) {
            std::vector<Cell> path = {cell};  // восстанавливаем путь от выхода к старту
            for (Cell p = cell; parent[p] != p; p = parent[p]) path.push_back(parent[p]);
            std::reverse(path.begin(), path.end());
            return path;
        }
        for (int m = 0; m < 4; ++m) {
            Cell next = {cell.first + DR[m], cell.second + DC[m]};
            if (maze[next.first][next.second] == 0 && parent.count(next) == 0) {
                parent[next] = cell;
                queue.push(next);
            }
        }
    }
    return {};
}

int main() {
    Grid maze;
    for (const char* line : MAZE) {
        maze.emplace_back();
        for (const char* p = line; *p != '\0'; ++p) maze.back().push_back(*p == '#');
    }
    std::vector<Cell> path = shortest_exit(maze, {5, 1});
    if (path.empty()) {
        std::cout << "Выхода нет\n";
        return 0;
    }
    std::cout << "Шагов: " << path.size() - 1 << "\n";
    for (size_t i = 0; i < path.size(); ++i) {
        if (i > 0) std::cout << ' ';
        std::cout << '(' << path[i].first << ',' << path[i].second << ')';
    }
    std::cout << "\n";
    return 0;
}

Реализация на Rust

// Поиск в ширину (волна на графе): кратчайший путь от клетки (row, col) до границы лабиринта, O(N*M) времени
// В лабиринте 0 - проходимая клетка, 1 - стена; выход - любая проходимая граничная клетка
use std::collections::{HashMap, VecDeque};

pub const DR: [i32; 4] = [0, 1, 0, -1];
pub const DC: [i32; 4] = [1, 0, -1, 0];
pub const MAZE: [&str; 7] = [
    "####.####", "#.......#", "#.#####.#", "#.#...#.#", "#.#.#.#.#", "#...#...#", "#########",
];

fn on_border(grid: &[Vec<u8>], row: usize, col: usize) -> bool {
    row == 0 || row == grid.len() - 1 || col == 0 || col == grid[0].len() - 1
}

// Восстанавливает путь от старта до клетки cell по ссылкам на предыдущие клетки
fn restore(parent: &HashMap<(usize, usize), (usize, usize)>, start: (usize, usize), cell: (usize, usize)) -> Vec<(usize, usize)> {
    let mut path = vec![cell];
    let mut current = cell;
    while current != start {
        current = parent[&current];
        path.push(current);
    }
    path.reverse();
    path
}

// Кратчайший путь от start до границы (список клеток) или None: клетки обходятся очередью
pub fn shortest_exit(maze: &[Vec<u8>], start: (usize, usize)) -> Option<Vec<(usize, usize)>> {
    if maze[start.0][start.1] == 1 {
        return None;
    }
    let mut parent = HashMap::from([(start, start)]);
    let mut queue = VecDeque::from([start]);
    while let Some((row, col)) = queue.pop_front() {
        if on_border(maze, row, col) {
            return Some(restore(&parent, start, (row, col)));
        }
        for m in 0..4 {
            let r = (row as i32 + DR[m]) as usize;
            let c = (col as i32 + DC[m]) as usize;
            if maze[r][c] == 0 && !parent.contains_key(&(r, c)) {
                parent.insert((r, c), (row, col));
                queue.push_back((r, c));
            }
        }
    }
    None
}

fn main() {
    let maze: Vec<Vec<u8>> = MAZE
        .iter()
        .map(|line| line.bytes().map(|ch| u8::from(ch == b'#')).collect())
        .collect();
    match shortest_exit(&maze, (5, 1)) {
        None => println!("Выхода нет"),
        Some(path) => {
            println!("Шагов: {}", path.len() - 1);
            let cells: Vec<String> = path.iter().map(|(r, c)| format!("({r},{c})")).collect();
            println!("{}", cells.join(" "));
        }
    }
}

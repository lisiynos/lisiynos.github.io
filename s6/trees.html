Обходы бинарного дерева

Бинарное дерево — это дерево, в котором у каждой вершины не более двух детей: левый и правый. Обход дерева — порядок, в котором посещаются все его вершины. Для бинарного дерева естественны три рекурсивных обхода, которые различаются тем, когда посещается корень относительно поддеревьев.

Обозначим корень через $V$, левое поддерево через $L$, правое через $R$. Прямой обход (pre-order) посещает сначала $V$, затем $L$, затем $R$. Он удобен, когда корень нужно обработать раньше потомков, например при копировании дерева.

Симметричный обход (in-order) посещает $L$, затем $V$, затем $R$. Для двоичного дерева поиска он выводит ключи в отсортированном порядке.

Обратный обход (post-order) посещает $L$, затем $R$, затем $V$. Корень обрабатывается, когда результаты для обоих поддеревьев уже известны, например при подсчёте размеров поддеревьев.

Каждая вершина посещается ровно один раз и обрабатывается за $O(1)$, поэтому время работы $O(n)$, где $n$ — число вершин. Глубина рекурсии равна высоте дерева $h$, поэтому память $O(h)$; в худшем случае $h = n$.

Реализация на Python

# Обходы бинарного дерева: прямой, симметричный, обратный. Сложность O(n) по времени, O(h) по памяти.

LEFT = [1, 3, -1, 6, -1, -1, -1]  # левый ребёнок вершины, -1 если его нет
RIGHT = [2, 4, 5, -1, -1, -1, -1]  # правый ребёнок вершины, -1 если его нет


def preorder(left, right, v, order):
    """Прямой обход: корень, затем левое и правое поддеревья."""
    if v == -1:
        return
    order.append(v)
    preorder(left, right, left[v], order)
    preorder(left, right, right[v], order)


def inorder(left, right, v, order):
    """Симметричный обход: левое поддерево, корень, правое поддерево."""
    if v == -1:
        return
    inorder(left, right, left[v], order)
    order.append(v)
    inorder(left, right, right[v], order)


def postorder(left, right, v, order):
    """Обратный обход: левое и правое поддеревья, затем корень."""
    if v == -1:
        return
    postorder(left, right, left[v], order)
    postorder(left, right, right[v], order)
    order.append(v)


def main():
    for name, walk in (("прямой", preorder), ("симметричный", inorder), ("обратный", postorder)):
        order = []
        walk(LEFT, RIGHT, 0, order)
        print(name, *order)


if __name__ == "__main__":
    main()

Реализация на C++

// Обходы бинарного дерева: прямой, симметричный, обратный. Сложность O(n) по времени, O(h) по памяти.
#include <iostream>
#include <vector>

// Прямой обход: корень, затем левое и правое поддеревья.
void preorder(const std::vector<int>& left, const std::vector<int>& right, int v, std::vector<int>& order) {
    if (v == -1) return;
    order.push_back(v);
    preorder(left, right, left[v], order);
    preorder(left, right, right[v], order);
}

// Симметричный обход: левое поддерево, корень, правое поддерево.
void inorder(const std::vector<int>& left, const std::vector<int>& right, int v, std::vector<int>& order) {
    if (v == -1) return;
    inorder(left, right, left[v], order);
    order.push_back(v);
    inorder(left, right, right[v], order);
}

// Обратный обход: левое и правое поддеревья, затем корень.
void postorder(const std::vector<int>& left, const std::vector<int>& right, int v, std::vector<int>& order) {
    if (v == -1) return;
    postorder(left, right, left[v], order);
    postorder(left, right, right[v], order);
    order.push_back(v);
}

// Печатает название обхода и вершины в порядке посещения.
void print_order(const char* name, const std::vector<int>& order) {
    std::cout << name << ':';
    for (int v : order) std::cout << ' ' << v;
    std::cout << '\n';
}

int main() {
    // Дерево из 7 вершин: left[v] и right[v] — дети вершины v, -1 если ребёнка нет.
    const std::vector<int> left = {1, 3, -1, 6, -1, -1, -1};
    const std::vector<int> right = {2, 4, 5, -1, -1, -1, -1};

    std::vector<int> order;
    preorder(left, right, 0, order);
    print_order("прямой", order);

    order.clear();
    inorder(left, right, 0, order);
    print_order("симметричный", order);

    order.clear();
    postorder(left, right, 0, order);
    print_order("обратный", order);
    return 0;
}

}

    if v == -1 {
        return;
    }
    let u = v as usize;
    inorder(left, right, left[u], order);
    order.push(v);
    inorder(left, right, right[u], order);
}

// Обратный обход: левое и правое поддеревья, затем корень.
pub fn postorder(left: &[i32], right: &[i32], v: i32, order: &mut Vec<i32>) {

Высота дерева и число листьев

Высота дерева $h$ — число вершин на самом длинном пути от корня до листа. Высота пустого дерева равна $0$, высота одной вершины — $1$. Лист — вершина, у которой нет детей.

Высота считается по рекуррентной формуле: $h(\varnothing) = 0$, а для непустого поддерева с корнем $v$ верно $h(v) = 1 + \max(h(L_v), h(R_v))$, где $L_v$ и $R_v$ — левый и правый ребёнок $v$. Сначала находим высоты детей, затем берём наибольшую и добавляем корень.

Число листьев считается так: $\mathrm{leaves}(\varnothing) = 0$, для листа $\mathrm{leaves}(v) = 1$, а в остальных случаях $\mathrm{leaves}(v) = \mathrm{leaves}(L_v) + \mathrm{leaves}(R_v)$. Вершина с одним ребёнком учитывается через пустое поддерево, которое даёт $0$.

Каждая из двух функций посещает каждую вершину ровно один раз, поэтому время работы $O(n)$. Глубина рекурсии равна высоте, значит память $O(h)$; для цепочки из $n$ вершин это $O(n)$.

Реализация на Python

# Высота дерева и число листьев, рекурсивно. Сложность O(n) по времени, O(h) по памяти.

LEFT = [1, 3, -1, 6, -1, -1, -1]  # левый ребёнок вершины, -1 если его нет
RIGHT = [2, 4, 5, -1, -1, -1, -1]  # правый ребёнок вершины, -1 если его нет


def height(left, right, v):
    """Число вершин на самом длинном пути от v вниз до листа; пустое дерево имеет высоту 0."""
    if v == -1:
        return 0
    return 1 + max(height(left, right, left[v]), height(left, right, right[v]))


def count_leaves(left, right, v):
    """Число листьев в поддереве вершины v; лист — вершина без детей."""
    if v == -1:
        return 0
    if left[v] == -1 and right[v] == -1:
        return 1
    return count_leaves(left, right, left[v]) + count_leaves(left, right, right[v])


def main():
    print("высота:", height(LEFT, RIGHT, 0))
    print("листья:", count_leaves(LEFT, RIGHT, 0))


if __name__ == "__main__":
    main()

Реализация на C++

// Высота дерева и число листьев, рекурсивно. Сложность O(n) по времени, O(h) по памяти.
#include <algorithm>
#include <iostream>
#include <vector>

// Высота поддерева вершины v: число вершин на самом длинном пути вниз до листа; пустое дерево имеет высоту 0.
int height(const std::vector<int>& left, const std::vector<int>& right, int v) {
    if (v == -1) return 0;
    return 1 + std::max(height(left, right, left[v]), height(left, right, right[v]));
}

// Число листьев в поддереве вершины v; лист — вершина без детей.
int count_leaves(const std::vector<int>& left, const std::vector<int>& right, int v) {
    if (v == -1) return 0;
    if (left[v] == -1 && right[v] == -1) return 1;
    return count_leaves(left, right, left[v]) + count_leaves(left, right, right[v]);
}

int main() {
    // Дерево из 7 вершин: left[v] и right[v] — дети вершины v, -1 если ребёнка нет.
    const std::vector<int> left = {1, 3, -1, 6, -1, -1, -1};
    const std::vector<int> right = {2, 4, 5, -1, -1, -1, -1};
    std::cout << "высота: " << height(left, right, 0) << '\n';
    std::cout << "листья: " << count_leaves(left, right, 0) << '\n';
    return 0;
}

Реализация на Rust


// Высота дерева и число листьев, рекурсивно. Сложность O(n) по времени, O(h) по памяти.

// Высота поддерева вершины v: число вершин на самом длинном пути вниз до листа; пустое дерево имеет высоту 0.
pub fn height(left: &[i32], right: &[i32], v: i32) -> usize {
    if v == -1 {
        return 0;
    }
    let u = v as usize;
    1 + height(left, right, left[u]).max(height(left, right, right[u]))
}

// Число листьев в поддереве вершины v; лист — вершина без детей.
pub fn count_leaves(left: &[i32], right: &[i32], v: i32) -> usize {
    if v == -1 {
        return 0;
    }
    let u = v as usize;
    if left[u] == -1 && right[u] == -1 {
        return 1;
    }
    count_leaves(left, right, left[u]) + count_leaves(left, right, right[u])
}

fn main() {
    // Дерево из 7 вершин: left[v] и right[v] — дети вершины v, -1 если ребёнка нет.
    let left = [1, 3, -1, 6, -1, -1, -1];
    let right = [2, 4, 5, -1, -1, -1, -1];
    println!("высота: {}", height(&left, &right, 0));
    println!("листья: {}", count_leaves(&left, &right, 0));
}

Наименьший общий предок

Наименьший общий предок (LCA, lowest common ancestor) вершин $u$ и $v$ — самая глубокая вершина, которая является предком и для $u$, и для $v$. Вершина считается предком самой себя: если $u$ лежит на пути от $v$ к корню, ответ равен $u$.

Дерево задано массивом родителей: $p[v]$ — родитель вершины $v$, у корня $p[v] = -1$. Глубина $d(v)$ — число рёбер от корня до $v$. Её можно найти подъёмом по массиву родителей за $O(h)$.

Сначала поднимаем более глубокую вершину на $|d(u) - d(v)|$ шагов, чтобы обе оказались на одной глубине. Если они совпали, это и есть ответ. Иначе поднимаем обе вершины на один шаг вверх и повторяем, пока они не совпадут.

Почему это верно: на одной глубине предки $u$ и $v$ совпадают тогда и только тогда, когда они являются общими предками. Значит, первое совпадение при подъёме снизу — самый глубокий общий предок.

Вычисление глубин и оба подъёма делают не более $h$ шагов каждый, поэтому время работы $O(h)$. Для сбалансированного дерева $h = O(\log n)$, для цепочки $h = O(n)$.

Реализация на Python

# Наименьший общий предок по массиву родителей. Сложность O(h) по времени, где h — высота дерева.

PARENT = [-1, 0, 0, 1, 1, 2, 3]  # родитель вершины, у корня -1


def depth(parent, v):
    """Глубина вершины v: число рёбер от корня до v."""
    steps = 0
    while parent[v] != -1:
        v = parent[v]
        steps += 1
    return steps


def lca(parent, u, v):
    """Поднимаем более глубокую вершину до уровня другой, затем поднимаем обе, пока не сойдутся."""
    depth_u = depth(parent, u)
    depth_v = depth(parent, v)
    while depth_u > depth_v:
        u = parent[u]
        depth_u -= 1
    while depth_v > depth_u:
        v = parent[v]
        depth_v -= 1
    while u != v:
        u = parent[u]
        v = parent[v]
    return u


def main():
    pairs = [(6, 4), (6, 5), (3, 4), (5, 6), (2, 5), (6, 6), (0, 6)]
    for u, v in pairs:
        print(f"lca({u}, {v}) = {lca(PARENT, u, v)}")


if __name__ == "__main__":
    main()

Реализация на C++

// Наименьший общий предок по массиву родителей. Сложность O(h) по времени, где h — высота дерева.
#include <iostream>
#include <utility>
#include <vector>

// Глубина вершины v: число рёбер от корня до v. Подъём по массиву родителей.
int depth(const std::vector<int>& parent, int v) {
    int steps = 0;
    while (parent[v] != -1) {
        v = parent[v];
        ++steps;
    }
    return steps;
}

// Поднимаем более глубокую вершину до уровня другой, затем поднимаем обе, пока не сойдутся.
int lca(const std::vector<int>& parent, int u, int v) {
    int depth_u = depth(parent, u);
    int depth_v = depth(parent, v);
    while (depth_u > depth_v) {
        u = parent[u];
        --depth_u;
    }
    while (depth_v > depth_u) {
        v = parent[v];
        --depth_v;
    }
    while (u != v) {
        u = parent[u];
        v = parent[v];
    }
    return u;
}

int main() {
    // Массив родителей: parent[v] — родитель вершины v, у корня -1.
    const std::vector<int> parent = {-1, 0, 0, 1, 1, 2, 3};
    const std::vector<std::pair<int, int>> pairs = {{6, 4}, {6, 5}, {3, 4}, {5, 6}, {2, 5}, {6, 6}, {0, 6}};
    for (auto [u, v] : pairs) {
        std::cout << "lca(" << u << ", " << v << ") = " << lca(parent, u, v) << '\n';
    }
    return 0;
}

Реализация на Rust


// Наименьший общий предок по массиву родителей. Сложность O(h) по времени, где h — высота дерева.

// Глубина вершины v: число рёбер от корня до v. Подъём по массиву родителей.
fn depth(parent: &[i32], mut v: i32) -> usize {
    let mut steps = 0;
    while parent[v as usize] != -1 {
        v = parent[v as usize];
        steps += 1;
    }
    steps
}

// Поднимаем более глубокую вершину до уровня другой, затем поднимаем обе, пока не сойдутся.
pub fn lca(parent: &[i32], mut u: i32, mut v: i32) -> i32 {
    let mut depth_u = depth(parent, u);
    let mut depth_v = depth(parent, v);
    while depth_u > depth_v {
        u = parent[u as usize];
        depth_u -= 1;
    }
    while depth_v > depth_u {
        v = parent[v as usize];
        depth_v -= 1;
    }
    while u != v {
        u = parent[u as usize];
        v = parent[v as usize];
    }
    u
}

fn main() {
    // Массив родителей: parent[v] — родитель вершины v, у корня -1.
    let parent = [-1, 0, 0, 1, 1, 2, 3];
    for (u, v) in [(6, 4), (6, 5), (3, 4), (5, 6), (2, 5), (6, 6), (0, 6)] {
        println!("lca({}, {}) = {}", u, v, lca(&parent, u, v));
    }
}

Стек вызовов функций

Каждый вызов функции записывается в стек вызовов: в кадре хранятся адрес возврата, аргументы и локальные переменные. Когда функция возвращает результат, её кадр снимается со стека. Рекурсивная функция вызывает саму себя, поэтому кадров на стеке накапливается столько, сколько уровней рекурсии. Отсюда следует, что у рекурсии обязательно должен быть базовый случай: без него стек переполняется.

Наибольший общий делитель: алгоритм Евклида

Наибольший общий делитель целых неотрицательных чисел $a$ и $b$ обозначим $\operatorname{gcd}(a, b)$. Алгоритм Евклида опирается на два равенства: $\operatorname{gcd}(a, 0) = a$ и $\operatorname{gcd}(a, b) = \operatorname{gcd}(b, a \bmod b)$ при $b \neq 0$.

Это рекурсия: функция вызывает себя с парой $(b, a \bmod b)$. Базовый случай $b = 0$ останавливает рекурсию, и ответом служит $a$.

Остаток $a \bmod b$ всегда меньше $b$, поэтому второй аргумент строго убывает. За два шага остаток уменьшается более чем вдвое, поэтому число шагов равно $O(\log \min(a, b))$.

Реализация на Python

# НОД по алгоритму Евклида (рекурсия): O(log min(a, b))
def gcd(a, b):
    """Возвращает наибольший общий делитель неотрицательных целых a и b.

    Использует gcd(a, b) = gcd(b, a mod b) и gcd(a, 0) = a.
    """
    # Базовый случай: остаток стал нулём, значит a и есть ответ
    if b == 0:
        return a
    # Переход к паре (b, a mod b): второй аргумент на каждом шаге уменьшается
    return gcd(b, a % b)


def main():
    print(f"НОД(48, 18) = {gcd(48, 18)}")
    print(f"НОД(17, 0) = {gcd(17, 0)}")
    print(f"НОД(0, 5) = {gcd(0, 5)}")
    print(f"НОД(0, 0) = {gcd(0, 0)}")


if __name__ == "__main__":
    main()

Реализация на C++

// НОД по алгоритму Евклида (рекурсия): O(log min(a, b))
#include <cstdio>

// Возвращает наибольший общий делитель неотрицательных целых a и b.
// Использует gcd(a, b) = gcd(b, a mod b) и gcd(a, 0) = a.
unsigned long long gcd(unsigned long long a, unsigned long long b) {
  // Базовый случай: остаток стал нулём, значит a и есть ответ
  if (b == 0) return a;
  // Переход к паре (b, a mod b): второй аргумент на каждом шаге уменьшается
  return gcd(b, a % b);
}

int main() {
  std::printf("НОД(48, 18) = %llu\n", gcd(48, 18));
  std::printf("НОД(17, 0) = %llu\n", gcd(17, 0));
  std::printf("НОД(0, 5) = %llu\n", gcd(0, 5));
  std::printf("НОД(0, 0) = %llu\n", gcd(0, 0));
  return 0;
}

Ханойские башни: рекурсивное перемещение

Есть три стержня $A$, $B$ и $C$ и $n$ дисков разного размера, надетых на стержень $A$ так, что меньший диск лежит выше большего. Нужно переложить все диски на стержень $C$. За один ход переносится верхний диск, и нельзя класть больший диск на меньший.

Рекурсивная идея такова: чтобы перенести $n$ дисков с $A$ на $C$, сначала переносим $n-1$ верхних дисков с $A$ на $B$, затем самый большой диск с $A$ на $C$, и наконец $n-1$ дисков с $B$ на $C$. Каждый из двух переносов $n-1$ дисков решается той же схемой, только стержни меняют роли.

Пусть $T(n)$ — число ходов. Тогда $T(n) = 2T(n-1) + 1$ и $T(0) = 0$, откуда $T(n) = 2^n - 1$. Сложность алгоритма $O(2^n)$: при $n = 20$ ходов уже больше миллиона, поэтому выводить все ходы имеет смысл только для небольших $n$.

Реализация на Python

# Ханойские башни: перенос n дисков со стержня src на dst через aux, O(2^n) ходов
def hanoi(n, src, dst, aux, moves):
    """Добавляет в moves все ходы (откуда, куда) для переноса n дисков.

    Сначала n-1 верхних дисков уходят на вспомогательный стержень aux,
    затем самый большой диск переносится на dst, и n-1 дисков идут с aux на dst.
    """
    # Дисков нет, переносить нечего: рекурсия останавливается
    if n == 0:
        return
    hanoi(n - 1, src, aux, dst, moves)
    moves.append((src, dst))
    hanoi(n - 1, aux, dst, src, moves)


def main():
    moves = []
    hanoi(3, "A", "C", "B", moves)
    for src, dst in moves:
        print(f"{src} -> {dst}")
    print(f"всего ходов: {len(moves)}")


if __name__ == "__main__":
    main()

Реализация на C++

// Ханойские башни: перенос n дисков со стержня src на dst через aux, O(2^n) ходов
#include <cstdio>
#include <utility>
#include <vector>

// Добавляет в moves все ходы (откуда, куда) для переноса n дисков.
// Сначала n-1 верхних дисков уходят на aux, затем большой диск на dst, и n-1 дисков идут с aux на dst.
void hanoi(int n, char src, char dst, char aux, std::vector<std::pair<char, char>>& moves) {
  // Дисков нет, переносить нечего: рекурсия останавливается
  if (n == 0) return;
  hanoi(n - 1, src, aux, dst, moves);
  moves.emplace_back(src, dst);
  hanoi(n - 1, aux, dst, src, moves);
}

int main() {
  std::vector<std::pair<char, char>> moves;
  hanoi(3, 'A', 'C', 'B', moves);
  for (const auto& [from, to] : moves) {
    std::printf("%c -> %c\n", from, to);
  }
  std::printf("всего ходов: %zu\n", moves.size());
  return 0;
}

Перебор с возвратом: все подмножества

Подмножество множества из $n$ элементов задаётся решениями для каждого элемента: брать его или не брать. Так получаются все $2^n$ подмножеств.

Перебор с возвратом — это рекурсия по номеру элемента. На каждом уровне сначала пробуем не брать элемент и уходим вглубь, затем берём его и снова уходим вглубь. Возврат из рекурсии отменяет выбор (последний элемент удаляется из текущего подмножества), и перебор продолжается в другой ветке.

Глубина рекурсии равна $n$, а в каждом из $2^n$ листьев дерева перебора копируется текущее подмножество длиной до $n$. Поэтому сложность равна $O(n \cdot 2^n)$.

Реализация на Python

# Перебор с возвратом: все подмножества множества из n элементов, O(n * 2^n)
def search(items, index, current, result):
    """Перебирает выбор элементов с позиции index: сначала не берём элемент, потом берём."""
    # Все элементы решены: сохраняем копию текущего подмножества
    if index == len(items):
        result.append(list(current))
        return
    search(items, index + 1, current, result)  # элемент не берём
    current.append(items[index])               # пробуем взять элемент
    search(items, index + 1, current, result)
    current.pop()                              # возврат: отменяем выбор


def subsets(items):
    """Возвращает список всех подмножеств списка items."""
    result = []
    search(items, 0, [], result)
    return result


def main():
    all_subsets = subsets(["a", "b", "c"])
    for subset in all_subsets:
        print(subset)
    print(f"всего подмножеств: {len(all_subsets)}")


if __name__ == "__main__":
    main()

Реализация на C++

// Перебор с возвратом: все подмножества множества из n элементов, O(n * 2^n)
#include <cstddef>
#include <cstdio>
#include <vector>

// Перебирает выбор элементов с позиции index: сначала не берём элемент, потом берём.
// На листе (все элементы решены) сохраняем копию текущего подмножества.
void search(const std::vector<int>& items, std::size_t index, std::vector<int>& current,
            std::vector<std::vector<int>>& result) {
  if (index == items.size()) {
    result.push_back(current);
    return;
  }
  search(items, index + 1, current, result);  // элемент не берём
  current.push_back(items[index]);            // пробуем взять элемент
  search(items, index + 1, current, result);
  current.pop_back();                         // возврат: отменяем выбор
}

std::vector<std::vector<int>> subsets(const std::vector<int>& items) {
  std::vector<std::vector<int>> result;
  std::vector<int> current;
  search(items, 0, current, result);
  return result;
}

int main() {
  std::vector<std::vector<int>> all = subsets({1, 2, 3});
  for (const auto& subset : all) {
    std::printf("{");
    for (std::size_t i = 0; i < subset.size(); ++i) {
      if (i > 0) std::printf(", ");
      std::printf("%d", subset[i]);
    }
    std::printf("}\n");
  }
  std::printf("всего подмножеств: %zu\n", all.size());
  return 0;
}

Реализация на Rust


// Перебор с возвратом: все подмножества множества из n элементов, O(n * 2^n)
// Перебирает выбор элементов с позиции index: сначала не берём элемент, потом берём.
// На листе (все элементы решены) сохраняем копию текущего подмножества.
fn search(items: &[i32], index: usize, current: &mut Vec<i32>, result: &mut Vec<Vec<i32>>) {
    if index == items.len() {
        result.push(current.clone());
        return;
    }
    search(items, index + 1, current, result); // элемент не берём
    current.push(items[index]); // пробуем взять элемент
    search(items, index + 1, current, result);
    current.pop(); // возврат: отменяем выбор
}

pub fn subsets(items: &[i32]) -> Vec<Vec<i32>> {
    let mut result = Vec::new();
    search(items, 0, &mut Vec::new(), &mut result);
    result
}

fn main() {
    let all = subsets(&[1, 2, 3]);
    for subset in &all {
        println!("{subset:?}");
    }
    println!("всего подмножеств: {}", all.len());
}

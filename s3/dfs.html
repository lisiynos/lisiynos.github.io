DFS - Поиск в глубину

Постановка задачи, рекурсивная реализация, отличия от BFS (поиска в ширину)

Реализация на Python


# Поиск в глубину (DFS), рекурсивный обход графа: O(V + E) при списках смежности


def dfs(adjacency, v, visited, order):
    """Рекурсивно посещает вершину v и все достижимые из неё непосещённые вершины."""
    if visited[v]:
        return
    visited[v] = True
    order.append(v)  # Запоминаем порядок посещения вершин
    for u in adjacency[v]:
        dfs(adjacency, u, visited, order)


def dfs_order(adjacency, source):
    """Возвращает порядок посещения вершин при обходе в глубину из source."""
    if not adjacency:
        return []
    visited = [False] * len(adjacency)
    order = []
    dfs(adjacency, source, visited, order)
    return order


if __name__ == "__main__":
    # Демонстрация: граф из 9 вершин (нумерация с нуля), тот же, что в примерах на странице
    adjacency = [
        [1, 7], [0, 2, 7], [1, 3, 7], [2, 6, 8], [5, 6],
        [4], [3, 4, 7], [0, 1, 2, 6], [3],
    ]
    # Ожидаемый порядок посещения: [0, 1, 2, 3, 6, 4, 5, 7, 8]
    print(dfs_order(adjacency, 0))

Реализация на Rust

// Поиск в глубину (DFS), рекурсивный обход графа: O(V + E) при списках смежности

/// Рекурсивно посещает вершину v и все достижимые из неё непосещённые вершины.
pub fn dfs(adjacency: &[Vec<usize>], v: usize, visited: &mut [bool], order: &mut Vec<usize>) {
    if visited[v] {
        return;
    }
    visited[v] = true;
    order.push(v); // Запоминаем порядок посещения вершин
    for &u in &adjacency[v] {
        dfs(adjacency, u, visited, order);
    }
}

/// Возвращает порядок посещения вершин при обходе в глубину из source.
pub fn dfs_order(adjacency: &[Vec<usize>], source: usize) -> Vec<usize> {
    let mut order = Vec::new();
    if adjacency.is_empty() {
        return order;
    }
    let mut visited = vec![false; adjacency.len()];
    dfs(adjacency, source, &mut visited, &mut order);
    order
}

fn main() {
    // Демонстрация: граф из 9 вершин (нумерация с нуля), тот же, что в примерах на странице
    let adjacency: Vec<Vec<usize>> = vec![
        vec![1, 7], vec![0, 2, 7], vec![1, 3, 7], vec![2, 6, 8], vec![5, 6],
        vec![4], vec![3, 4, 7], vec![0, 1, 2, 6], vec![3],
    ];
    // Ожидаемый порядок посещения: 0 1 2 3 6 4 5 7 8
    let order = dfs_order(&adjacency, 0);
    let line: Vec<String> = order.iter().map(|v| v.to_string()).collect();
    println!("{}", line.join(" "));
}

Анализ времени работы для различных способов задания графа

Матрица смежности

Самый прямой способ хранить граф на $V$ вершинах — матрица смежности: квадратная таблица $A$ размера $V \times V$, в которой $A[u][v] = 1$, если в графе есть ребро $(u, v)$, и $A[u][v] = 0$ в противном случае. Для неориентированного графа матрица симметрична: ребро $(u, v)$ записывается и в $A[u][v]$, и в $A[v][u]$. Для взвешенного графа в ячейку записывают вес ребра, а отсутствие ребра отмечают особым значением.

Матрица занимает $V^2$ ячеек независимо от числа рёбер $E$, поэтому память составляет $O(V^2)$. Для разреженного графа это расточительно: при $V = 10^5$ матрица содержит $10^{10}$ ячеек, даже если рёбер всего $10^5$.

Основные операции с матрицей простые. Проверка наличия ребра $(u, v)$, его добавление и удаление обращаются к одной ячейке (для неориентированного графа — к двум симметричным) и выполняются за $O(1)$. Перебор соседей вершины $v$ — это просмотр всей строки $A[v]$ из $V$ ячеек, поэтому он занимает $O(V)$ независимо от числа соседей.

Обход в глубину на матрице смежности устроен так: при посещении вершины $v$ перебираются все номера $u$ от $0$ до $V-1$; для каждого $u$, для которого $A[v][u] = 1$ и вершина $u$ ещё не посещена, рекурсивно запускается обход из $u$. Ниже граф на 9 вершинах (нумерация с нуля) задаётся списком рёбер, по нему строится матрица, а затем выводится порядок посещения вершин. Это тот же граф, что и в примерах выше.

Реализация на Python

# Поиск в глубину (DFS) на матрице смежности: O(V^2) по времени, O(V^2) по памяти


def build_adjacency_matrix(vertices, edges):
    """Строит матрицу смежности неориентированного графа: adjacency_matrix[u][v] == 1, если есть ребро (u, v)."""
    adjacency_matrix = [[0] * vertices for _ in range(vertices)]
    for u, v in edges:
        adjacency_matrix[u][v] = 1
        adjacency_matrix[v][u] = 1  # Граф неориентированный: ребро видно из обоих концов
    return adjacency_matrix


def dfs(adjacency_matrix, v, visited, order):
    """Рекурсивно посещает вершину v и все достижимые из неё непосещённые вершины."""
    visited[v] = True
    order.append(v)  # Запоминаем порядок посещения вершин
    # Соседей ищем перебором всей строки v: поэтому на каждую вершину уходит O(V) времени
    for u in range(len(adjacency_matrix)):
        if adjacency_matrix[v][u] and not visited[u]:
            dfs(adjacency_matrix, u, visited, order)


def dfs_order(adjacency_matrix, source):
    """Возвращает порядок посещения вершин при обходе в глубину из source."""
    visited = [False] * len(adjacency_matrix)
    order = []
    dfs(adjacency_matrix, source, visited, order)
    return order

    # Демонстрация: граф из 9 вершин (нумерация с нуля), тот же, что в примерах выше
    edges = [(0, 1), (0, 7), (1, 2), (1, 7), (2, 3), (2, 7), (3, 6), (3, 8), (4, 5), (4, 6), (6, 7)]
    adjacency_matrix = build_adjacency_matrix(9, edges)
    # Ожидаемый порядок посещения: [0, 1, 2, 3, 6, 4, 5, 7, 8]
    print(dfs_order(adjacency_matrix, 0))

Реализация на C++

// Поиск в глубину (DFS) на матрице смежности: O(V^2) по времени, O(V^2) по памяти
#include <iostream>
#include <utility>
#include <vector>

using Matrix = std::vector<std::vector<int>>;

// Строит матрицу смежности неориентированного графа на n вершинах по списку рёбер
Matrix build_adjacency_matrix(int n, const std::vector<std::pair<int, int>>& edges) {
    Matrix adjacency_matrix(n, std::vector<int>(n, 0));
    for (const auto& [u, v] : edges) {
        adjacency_matrix[u][v] = 1;
        adjacency_matrix[v][u] = 1;  // Граф неориентированный: ребро видно из обоих концов
    }
    return adjacency_matrix;
}

// Рекурсивно посещает вершину v и все достижимые из неё непосещённые вершины
void dfs(const Matrix& adjacency_matrix, int v, std::vector<bool>& visited, std::vector<int>& order) {
    visited[v] = true;
    order.push_back(v);  // Запоминаем порядок посещения вершин
    // Соседей ищем перебором всей строки v: поэтому на каждую вершину уходит O(V) времени
    const int vertices = static_cast<int>(adjacency_matrix.size());
    for (int u = 0; u < vertices; ++u) {
        if (adjacency_matrix[v][u] && !visited[u]) dfs(adjacency_matrix, u, visited, order);
    }
}

// Возвращает порядок посещения вершин при обходе в глубину из source
std::vector<int> dfs_order(const Matrix& adjacency_matrix, int source) {
    std::vector<int> order;
    std::vector<bool> visited(adjacency_matrix.size(), false);
    dfs(adjacency_matrix, source, visited, order);
    return order;
}

int main() {
    // Демонстрация: граф из 9 вершин (нумерация с нуля), тот же, что в примерах выше
    const std::vector<std::pair<int, int>> edges = {
        {0, 1}, {0, 7}, {1, 2}, {1, 7}, {2, 3}, {2, 7}, {3, 6}, {3, 8}, {4, 5}, {4, 6}, {6, 7},
    };
    const Matrix adjacency_matrix = build_adjacency_matrix(9, edges);
    // Ожидаемый порядок посещения: 0 1 2 3 6 4 5 7 8
    for (const int v : dfs_order(adjacency_matrix, 0)) std::cout << v << ' ';
    std::cout << '\n';
    return 0;
}

Каждая вершина посещается не более одного раза, и при посещении просматривается строка длиной $V$. Всего выполняется не более $V \cdot V = V^2$ проверок, поэтому обход в глубину на матрице смежности работает за $O(V^2)$ независимо от числа рёбер. Дополнительно нужна память $O(V)$ под массив посещённых вершин и стек рекурсии.

Списки смежности

Другой способ — списки смежности: для каждой вершины $v$ хранится список её соседей, то есть номеров вершин, связанных с $v$ ребром. Для неориентированного графа каждое ребро $(u, v)$ записывается дважды: в список $u$ и в список $v$. Списков $V$, а суммарная длина всех списков равна $2E$, поэтому память составляет $O(V + E)$. Для разреженного графа это намного меньше $O(V^2)$ у матрицы.

Проверка наличия ребра $(u, v)$ требует просмотра списка вершины $u$ и занимает $O(\deg u)$, где $\deg u$ — число соседей вершины $u$. Добавление ребра — это дописывание номера в конец двух списков, поэтому оно занимает $O(1)$ (амортизированно для динамических массивов). Удаление ребра требует поиска в списках обоих концов и занимает $O(\deg u + \deg v)$. Перебор соседей вершины $v$ занимает $O(\deg v)$ и не зависит от числа вершин графа.

Обход в глубину на списках смежности устроен так же, но соседи берутся только из списка посещаемой вершины: для каждого $u$ из списка вершины $v$ рекурсивно запускается обход, если вершина $u$ ещё не посещена. Ниже те же рёбра строятся в списки, и обход выводит порядок посещения вершин.

Реализация на Python

# Поиск в глубину (DFS) на списках смежности: O(V + E) по времени, O(V + E) по памяти


def build_adjacency_lists(vertices, edges):
    """Строит списки смежности: adjacency_lists[v] содержит соседей вершины v в порядке добавления рёбер."""
    adjacency_lists = [[] for _ in range(vertices)]
    for u, v in edges:
        adjacency_lists[u].append(v)
        adjacency_lists[v].append(u)  # Граф неориентированный: ребро хранится в обоих списках
    return adjacency_lists


def dfs(adjacency_lists, v, visited, order):
    """Рекурсивно посещает вершину v и все достижимые из неё непосещённые вершины."""
    visited[v] = True
    order.append(v)  # Запоминаем порядок посещения вершин
    # Соседей берём из списка вершины v: на него уходит O(deg v) времени
    for u in adjacency_lists[v]:
        if not visited[u]:
            dfs(adjacency_lists, u, visited, order)


def dfs_order(adjacency_lists, source):
    """Возвращает порядок посещения вершин при обходе в глубину из source."""
    visited = [False] * len(adjacency_lists)
    order = []
    dfs(adjacency_lists, source, visited, order)
    return order

    # Демонстрация: тот же граф из 9 вершин, что и в варианте с матрицей
    edges = [(0, 1), (0, 7), (1, 2), (1, 7), (2, 3), (2, 7), (3, 6), (3, 8), (4, 5), (4, 6), (6, 7)]
    adjacency_lists = build_adjacency_lists(9, edges)
    # Ожидаемый порядок посещения: [0, 1, 2, 3, 6, 4, 5, 7, 8]
    print(dfs_order(adjacency_lists, 0))

Реализация на C++

// Поиск в глубину (DFS) на списках смежности: O(V + E) по времени, O(V + E) по памяти
#include <iostream>
#include <utility>
#include <vector>

using Lists = std::vector<std::vector<int>>;

// Строит списки смежности неориентированного графа на n вершинах по списку рёбер
Lists build_adjacency_lists(int n, const std::vector<std::pair<int, int>>& edges) {
    Lists adjacency_lists(n);
    for (const auto& [u, v] : edges) {
        adjacency_lists[u].push_back(v);
        adjacency_lists[v].push_back(u);  // Граф неориентированный: ребро хранится в обоих списках
    }
    return adjacency_lists;
}

// Рекурсивно посещает вершину v и все достижимые из неё непосещённые вершины
void dfs(const Lists& adjacency_lists, int v, std::vector<bool>& visited, std::vector<int>& order) {
    visited[v] = true;
    order.push_back(v);  // Запоминаем порядок посещения вершин
    // Соседей берём из списка вершины v: на него уходит O(deg v) времени
    for (const int u : adjacency_lists[v]) {
        if (!visited[u]) dfs(adjacency_lists, u, visited, order);
    }
}

// Возвращает порядок посещения вершин при обходе в глубину из source
std::vector<int> dfs_order(const Lists& adjacency_lists, int source) {
    std::vector<int> order;
    std::vector<bool> visited(adjacency_lists.size(), false);
    dfs(adjacency_lists, source, visited, order);
    return order;
}

int main() {
    // Демонстрация: тот же граф из 9 вершин, что и в варианте с матрицей
    const std::vector<std::pair<int, int>> edges = {
        {0, 1}, {0, 7}, {1, 2}, {1, 7}, {2, 3}, {2, 7}, {3, 6}, {3, 8}, {4, 5}, {4, 6}, {6, 7},
    };
    const Lists adjacency_lists = build_adjacency_lists(9, edges);
    // Ожидаемый порядок посещения: 0 1 2 3 6 4 5 7 8
    for (const int v : dfs_order(adjacency_lists, 0)) std::cout << v << ' ';
    std::cout << '\n';
    return 0;
}

Каждая вершина посещается не более одного раза, и при посещении $v$ просматривается её список длиной $\deg v$. Сумма степеней всех вершин равна $2E$, поэтому обход работает за $O(V + E)$. Для разреженного графа, где $E$ имеет порядок $V$, это $O(V)$, тогда как у варианта с матрицей обход занимает $O(V^2)$.

Порядок посещения вершин в этом варианте зависит от порядка рёбер во входных данных, потому что соседи берутся из списков в порядке добавления. В варианте с матрицей соседи перебираются по возрастанию номеров. Поэтому порядок обхода может различаться, но множество вершин, достижимых из стартовой, в обоих вариантах одинаковое.

Сравнение двух представлений для неориентированного графа с $V$ вершинами и $E$ рёбрами:

ОперацияМатрица смежностиСписки смежности
Память$O(V^2)$$O(V + E)$
Проверка ребра $(u, v)$$O(1)$$O(\deg u)$
Добавление ребра$O(1)$$O(1)$
Удаление ребра$O(1)$$O(\deg u + \deg v)$
Перебор соседей вершины $v$$O(V)$$O(\deg v)$
Обход в глубину (DFS)$O(V^2)$$O(V + E)$

Если граф плотный, то есть $E$ имеет порядок $V^2$, обе структуры занимают $O(V^2)$, а обход в глубину работает за $O(V^2)$, поэтому по порядку величины выигрыша нет. Для разреженных графов обычно выбирают списки смежности. Матрица удобна, когда граф небольшой, а проверка наличия ребра должна выполняться за $O(1)$.

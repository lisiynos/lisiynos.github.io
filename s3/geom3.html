Геометрия: построение выпуклой оболочки

  • Геометрия

Выпуклой оболочкой множества точек на плоскости является выпуклый многоугольник с вершинами из точек заданного множества. Существует несколько алгоритмов построения выпуклой оболочки. Самыми простыми для реализации являются алгоритмы Джарвиса и Грэхема.

Идея алгоритма Джарвиса построения выпуклой оболочки (его называют еще алгоритмом «заворачивания подарка») заключается в следующем:

  1. Ищется точка с минимальной координатой по оси $OX$. От нее задается первоначальное вертикальное направление.
  2. По очереди, перебирая все остальные точки, ищется такая, чтобы вектор с концом в этой точке, а  началом в исходной образовал с первоначальным вектором минимальный угол. Найденная таким образом точка будет точкой из выпуклой оболочки.
  3. Найденная точка принимается за исходную, новый построенный вектор задает исходное направление и из оставшихся $N-2$ точек ищется следующая точка.
  4. Процесс продолжается до первоначальной выбранной точки.

Реализация на Python

# Алгоритм Джарвиса (заворачивание подарка): O(N*H), H - число вершин оболочки
Point = tuple[int, int]


def cross(origin: Point, a: Point, b: Point) -> int:
    """Векторное произведение (a - origin) x (b - origin): > 0 - поворот влево."""
    return (a[0] - origin[0]) * (b[1] - origin[1]) - (a[1] - origin[1]) * (b[0] - origin[0])


def dist2(a: Point, b: Point) -> int:
    """Квадрат расстояния между точками."""
    return (a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2


def jarvis_hull(points: list[Point]) -> list[Point]:
    """Вершины выпуклой оболочки против часовой стрелки, начиная с самой левой точки."""
    unique = sorted(set(points))
    if len(unique) < 3:
        return unique
    start = unique[0]  # самая левая (при равенстве - самая нижняя) точка
    hull = []
    current = start
    while True:
        hull.append(current)
        # Ищем точку, относительно которой все остальные лежат слева от луча
        candidate = unique[1] if current == start else start
        for other in unique:
            if other == current:
                continue
            turn = cross(current, candidate, other)
            # Точка справа от луча (или на нем, но дальше) становится новым кандидатом
            if turn < 0 or (turn == 0 and dist2(current, other) > dist2(current, candidate)):
                candidate = other
        current = candidate
        if current == start:
            return hull


if __name__ == "__main__":
    demo_points = [(0, 0), (4, 0), (4, 4), (0, 4), (2, 2), (1, 3), (2, 0), (0, 2)]
    print("Оболочка:", jarvis_hull(demo_points))

Реализация на Rust


// Алгоритм Джарвиса (заворачивание подарка): O(N*H), H - число вершин оболочки
type Point = (i64, i64);

/// Векторное произведение (a - o) x (b - o): > 0 - поворот влево.
fn cross(o: Point, a: Point, b: Point) -> i64 {
    (a.0 - o.0) * (b.1 - o.1) - (a.1 - o.1) * (b.0 - o.0)
}

/// Квадрат расстояния между точками.
fn dist2(a: Point, b: Point) -> i64 {
    (a.0 - b.0).pow(2) + (a.1 - b.1).pow(2)
}

/// Вершины выпуклой оболочки против часовой стрелки, начиная с самой левой точки.
pub fn jarvis_hull(points: &[Point]) -> Vec<Point> {
    let mut unique = points.to_vec();
    unique.sort();
    unique.dedup();
    if unique.len() < 3 {
        return unique;
    }
    let start = unique[0]; // самая левая (при равенстве - самая нижняя) точка
    let mut hull = Vec::new();
    let mut current = start;
    loop {
        hull.push(current);
        // Ищем точку, относительно которой все остальные лежат слева от луча
        let mut candidate = if current == start { unique[1] } else { start };
        for &other in &unique {
            if other == current {
                continue;
            }
            let turn = cross(current, candidate, other);
            // Точка справа от луча (или на нем, но дальше) становится новым кандидатом
            if turn < 0 || (turn == 0 && dist2(current, other) > dist2(current, candidate)) {
                candidate = other;
            }
        }
        current = candidate;
        if current == start {
            return hull;
        }
    }
}

fn main() {
    let demo = [(0, 0), (4, 0), (4, 4), (0, 4), (2, 2), (1, 3), (2, 0), (0, 2)];
    println!("Оболочка: {:?}", jarvis_hull(&demo));
}

Реализация на C++

// Алгоритм Джарвиса (заворачивание подарка): O(N*H), H - число вершин оболочки
#include <algorithm>
#include <cstdio>
#include <utility>
#include <vector>

using Point = std::pair<long long, long long>;

// Векторное произведение (a - o) x (b - o): > 0 - поворот влево.
long long cross(Point o, Point a, Point b) {
    return (a.first - o.first) * (b.second - o.second) - (a.second - o.second) * (b.first - o.first);
}

// Квадрат расстояния между точками.
long long dist2(Point a, Point b) {
    return (a.first - b.first) * (a.first - b.first) + (a.second - b.second) * (a.second - b.second);
}

// Вершины выпуклой оболочки против часовой стрелки, начиная с самой левой точки.
std::vector<Point> jarvisHull(std::vector<Point> points) {
    std::sort(points.begin(), points.end());
    points.erase(std::unique(points.begin(), points.end()), points.end());
    if (points.size() < 3) return points;
    const Point start = points[0];  // самая левая (при равенстве - самая нижняя) точка
    std::vector<Point> hull;
    Point current = start;
    while (true) {
        hull.push_back(current);
        // Ищем точку, относительно которой все остальные лежат слева от луча
        Point candidate = (current == start) ? points[1] : start;
        for (const Point& other : points) {
            if (other == current) continue;
            long long turn = cross(current, candidate, other);
            // Точка справа от луча (или на нем, но дальше) становится новым кандидатом
            if (turn < 0 || (turn == 0 && dist2(current, other) > dist2(current, candidate))) {
                candidate = other;
            }
        }
        current = candidate;
        if (current == start) return hull;
    }
}

int main() {
    std::vector<Point> demo = {{0, 0}, {4, 0}, {4, 4}, {0, 4}, {2, 2}, {1, 3}, {2, 0}, {0, 2}};
    std::printf("Оболочка:");
    for (const Point& p : jarvisHull(demo)) {
        std::printf(" (%lld, %lld)", p.first, p.second);
    }
    std::printf("\n");
    return 0;
}

Идея алгоритма Грэхема построения выпуклой оболочки  заключается в следующем:

  1. Найти левую верхнюю точку. Запомнить ее.
  2. Отсортировать все остальные точки по возрастанию угла в полярной системе координат с центром в запомненной точке.
  3. По очереди перебирая все отсортированные точки, исключать те, которые будут содержаться внутри выпуклой оболочки. Оставшиеся при таком отборе точки и  составят выпуклую оболочку.

Реализация на Python

# Алгоритм Грэхема (сканирование по полярному углу): O(N*log N) на сортировку
from functools import cmp_to_key

Point = tuple[int, int]


def cross(origin: Point, a: Point, b: Point) -> int:
    """Векторное произведение (a - origin) x (b - origin): > 0 - поворот влево."""
    return (a[0] - origin[0]) * (b[1] - origin[1]) - (a[1] - origin[1]) * (b[0] - origin[0])


def dist2(a: Point, b: Point) -> int:
    """Квадрат расстояния между точками."""
    return (a[0] - b[0]) ** 2 + (a[1] - b[1]) ** 2


def graham_hull(points: list[Point]) -> list[Point]:
    """Вершины выпуклой оболочки против часовой стрелки, начиная с левой верхней точки."""
    unique = sorted(set(points))
    if len(unique) < 3:
        return unique
    # Опорная точка: самая верхняя, а при равенстве - самая левая
    pivot = max(unique, key=lambda p: (p[1], -p[0]))
    others = [p for p in unique if p != pivot]

    def by_polar_angle(a: Point, b: Point) -> int:
        turn = cross(pivot, a, b)
        if turn != 0:
            return -1 if turn > 0 else 1  # меньший полярный угол идет раньше
        return dist2(pivot, a) - dist2(pivot, b)  # на одном луче раньше ближняя точка

    others.sort(key=cmp_to_key(by_polar_angle))
    hull = [pivot]
    for point in others:
        # Если поворот не влево, предыдущая вершина лежит внутри оболочки - удаляем ее
        while len(hull) >= 2 and cross(hull[-2], hull[-1], point) <= 0:
            hull.pop()
        hull.append(point)
    return hull


if __name__ == "__main__":
    demo_points = [(0, 0), (4, 0), (4, 4), (0, 4), (2, 2), (1, 3), (2, 0), (0, 2)]
    print("Оболочка:", graham_hull(demo_points))

Реализация на Rust


// Алгоритм Грэхема (сканирование по полярному углу): O(N*log N) на сортировку
use std::cmp::Ordering;

type Point = (i64, i64);

/// Векторное произведение (a - o) x (b - o): > 0 - поворот влево.
fn cross(o: Point, a: Point, b: Point) -> i64 {
    (a.0 - o.0) * (b.1 - o.1) - (a.1 - o.1) * (b.0 - o.0)
}

/// Квадрат расстояния между точками.
fn dist2(a: Point, b: Point) -> i64 {
    (a.0 - b.0).pow(2) + (a.1 - b.1).pow(2)
}

/// Вершины выпуклой оболочки против часовой стрелки, начиная с левой верхней точки.
pub fn graham_hull(points: &[Point]) -> Vec<Point> {
    let mut unique = points.to_vec();
    unique.sort();
    unique.dedup();
    if unique.len() < 3 {
        return unique;
    }
    // Опорная точка: самая верхняя, а при равенстве - самая левая
    let pivot = *unique.iter().max_by_key(|p| (p.1, -p.0)).unwrap();
    let mut others: Vec<Point> = unique.iter().copied().filter(|&p| p != pivot).collect();
    // По возрастанию полярного угла; на одном луче раньше идет ближняя точка
    others.sort_by(|&a, &b| {
        let turn = cross(pivot, a, b);
        if turn != 0 {
            return if turn > 0 { Ordering::Less } else { Ordering::Greater };
        }
        dist2(pivot, a).cmp(&dist2(pivot, b))
    });
    let mut hull = vec![pivot];
    for p in others {
        // Если поворот не влево, предыдущая вершина лежит внутри оболочки - удаляем ее
        while hull.len() >= 2 && cross(hull[hull.len() - 2], hull[hull.len() - 1], p) <= 0 {
            hull.pop();
        }
        hull.push(p);
    }
    hull
}

fn main() {
    let demo = [(0, 0), (4, 0), (4, 4), (0, 4), (2, 2), (1, 3), (2, 0), (0, 2)];
    println!("Оболочка: {:?}", graham_hull(&demo));
}

Реализация на C++

// Алгоритм Грэхема (сканирование по полярному углу): O(N*log N) на сортировку
#include <algorithm>
#include <cstdio>
#include <utility>
#include <vector>

using Point = std::pair<long long, long long>;

// Векторное произведение (a - o) x (b - o): > 0 - поворот влево.
long long cross(Point o, Point a, Point b) {
    return (a.first - o.first) * (b.second - o.second) - (a.second - o.second) * (b.first - o.first);
}

// Квадрат расстояния между точками.
long long dist2(Point a, Point b) {
    return (a.first - b.first) * (a.first - b.first) + (a.second - b.second) * (a.second - b.second);
}

// Вершины выпуклой оболочки против часовой стрелки, начиная с левой верхней точки.
std::vector<Point> grahamHull(std::vector<Point> points) {
    std::sort(points.begin(), points.end());
    points.erase(std::unique(points.begin(), points.end()), points.end());
    if (points.size() < 3) return points;
    // Опорная точка: самая верхняя, а при равенстве - самая левая
    Point pivot = points[0];
    for (const Point& p : points) {
        if (p.second > pivot.second || (p.second == pivot.second && p.first < pivot.first)) pivot = p;
    }
    std::vector<Point> others;
    for (const Point& p : points) {
        if (p != pivot) others.push_back(p);
    }
    // По возрастанию полярного угла; на одном луче раньше идет ближняя точка
    std::sort(others.begin(), others.end(), [pivot](const Point& a, const Point& b) {
        long long turn = cross(pivot, a, b);
        if (turn != 0) return turn > 0;
        return dist2(pivot, a) < dist2(pivot, b);
    });
    std::vector<Point> hull = {pivot};
    for (const Point& p : others) {
        // Если поворот не влево, предыдущая вершина лежит внутри оболочки - удаляем ее
        while (hull.size() >= 2 && cross(hull[hull.size() - 2], hull.back(), p) <= 0) {
            hull.pop_back();
        }
        hull.push_back(p);
    }
    return hull;
}

int main() {
    std::vector<Point> demo = {{0, 0}, {4, 0}, {4, 4}, {0, 4}, {2, 2}, {1, 3}, {2, 0}, {0, 2}};
    std::printf("Оболочка:");
    for (const Point& p : grahamHull(demo)) {
        std::printf(" (%lld, %lld)", p.first, p.second);
    }
    std::printf("\n");
    return 0;
}

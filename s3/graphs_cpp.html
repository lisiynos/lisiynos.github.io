Количество простых путей

Простой путь - это путь на клеточном поле, который проходит через каждую клетку не более одного раза.

Нужно вычислить количество возможных простых путей из клетки с координатами $(0,0)$ длины $K$.

#include <cstdlib>
#include <iostream>
#include <cstdio>
#include <string>
#include <vector>
#include <algorithm>
using namespace std;

bool used[55][55]; // Клетка использована

long long Answer = 0;
int K;

void R(int x, int y, int t) {
  if (used[x][y]) return;
  if (t == K) { Answer++; return; }
  used[x][y] = true; // visit - посещаем вершину
  // Идём во все стороны
  R(x + 1, y, t + 1); R(x, y + 1, t + 1);     
  R(x - 1, y, t + 1); R(x, y - 1, t + 1);    
  used[x][y] = false; // leave - выходим из вершины
}

#define task  "paths"
int main() {
  freopen(task".in", "r", stdin);
  freopen(task".out", "w", stdout);
  cin >> K;
  for(int i = 0; i < 50; i++)
    for(int j = 0; j < 50; j++) 
      used[i][j] = false;
  R(20, 20, 0);
  cout << Answer;
}

Реализация на Python

# Подсчёт простых путей длины K из клетки (0, 0): перебор с возвратом, время O(4^K)
DIRECTIONS = ((1, 0), (0, 1), (-1, 0), (0, -1))


def count_paths(x, y, steps_left, visited):
    """Сколько путей длины steps_left из клетки (x, y) не проходят ни по одной клетке дважды."""
    if steps_left == 0:
        return 1
    visited.add((x, y))  # Клетка использована
    total = 0
    for dx, dy in DIRECTIONS:
        nx, ny = x + dx, y + dy
        if (nx, ny) not in visited:
            total += count_paths(nx, ny, steps_left - 1, visited)
    visited.remove((x, y))  # Выходим из клетки (откат)
    return total


def count_simple_paths(k):
    """Число простых путей длины k из клетки (0, 0)."""
    return count_paths(0, 0, k, set())


if __name__ == "__main__":
    # Демонстрация: количество простых путей длины 0..6 из клетки (0, 0)
    print([count_simple_paths(k) for k in range(7)])

Реализация на Rust

// Подсчёт простых путей длины K из клетки (0, 0): перебор с возвратом, время O(4^K)
use std::collections::HashSet;

pub const DIRECTIONS: [(i64, i64); 4] = [(1, 0), (0, 1), (-1, 0), (0, -1)];

/// Сколько путей длины steps_left из клетки (x, y) не проходят ни по одной клетке дважды.
fn count_paths(x: i64, y: i64, steps_left: u32, visited: &mut HashSet<(i64, i64)>) -> u64 {
    if steps_left == 0 {
        return 1;
    }
    visited.insert((x, y)); // Клетка использована
    let mut total = 0;
    for (dx, dy) in DIRECTIONS {
        let next = (x + dx, y + dy);
        if !visited.contains(&next) {
            total += count_paths(next.0, next.1, steps_left - 1, visited);
        }
    }
    visited.remove(&(x, y)); // Выходим из клетки (откат)
    total
}

/// Число простых путей длины k из клетки (0, 0).
pub fn count_simple_paths(k: u32) -> u64 {
    count_paths(0, 0, k, &mut HashSet::new())
}

fn main() {
    // Демонстрация: количество простых путей длины 0..6 из клетки (0, 0)
    let counts: Vec<u64> = (0..7).map(count_simple_paths).collect();
    println!("{counts:?}");
}
